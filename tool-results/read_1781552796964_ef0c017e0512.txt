     1→"use client";
     2→
     3→import { useState, useEffect } from "react";
     4→import { motion } from "framer-motion";
     5→import Link from "next/link";
     6→import { useParams } from "next/navigation";
     7→import { useLanguage } from "@/lib/i18n";
     8→import { Button } from "@/components/ui/button";
     9→import { Badge } from "@/components/ui/badge";
    10→import { Card, CardContent } from "@/components/ui/card";
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
    12→import { 
    13→  Globe, Moon, Sun, ChevronRight, ArrowLeft,
    14→  Clock, Play, Lock, BookOpen, Target, Lightbulb, 
    15→  FileText, HelpCircle, Beaker, CheckCircle, FlaskConical
    16→} from "lucide-react";
    17→import { loadStaticData } from "@/lib/static-data";
    18→import { getSimulationsByLessonId, getSimulationsBySubject, Simulation } from "@/lib/simulations";

    20→import { InteractiveQuiz } from "@/components/quiz/InteractiveQuiz";
    21→

    24→  textAr: string;

    28→

    39→  id: string;

    41→  explanationAr: string;
    42→  explanationEn: string;
    43→  order: number;

    45→
    46→interface Example {
    47→  id: string;
    48→  questionAr: string;
    49→  questionEn: string;
    50→  solutionAr: string;
    51→  solutionEn: string;
    52→  stepsAr: string;
    53→  stepsEn: string;
    54→  order: number;
    55→}
    56→

    59→  type: string;
    60→  questionAr: string;
    61→  questionEn: string;
    62→  optionsAr?: string;
    63→  optionsEn?: string;
    64→  answer: string;
    65→  explanationAr?: string;
    66→  explanationEn?: string;
    67→  points: number;
    68→  difficulty: string;

    71→

    79→  duration: number;
    80→  isFree: boolean;
    81→  videoUrl?: string;

    83→  introductionAr: string;
    84→  introductionEn: string;

    92→  Unit: {

   107→
   108→export default function LessonPage() {
   109→  const params = useParams();
   110→  const lessonId = params.id as string;
   111→  const { language, toggleLanguage, t } = useLanguage();
   112→  const [lesson, setLesson] = useState<LessonDetail | null>(null);
   113→  const [loading, setLoading] = useState(true);
   114→  const [isDark, setIsDark] = useState(false);
   115→  const [relatedSimulations, setRelatedSimulations] = useState<Simulation[]>([]);
   116→  const [showQuiz, setShowQuiz] = useState(false);
   117→
   118→  const isRTL = language === "ar";
   119→
   120→  useEffect(() => {
   121→    const fetchLesson = async () => {
   122→      try {
   123→        const data = await loadStaticData();
   124→        
   125→        let foundLesson: LessonDetail | null = null;
   126→        
   127→        for (const year of data?.academicYears || []) {
   128→          for (const subject of year.Subject || []) {
   129→            for (const unit of subject.Unit || []) {
   130→              const lesson = unit.Lesson?.find((l: any) => l.id === lessonId);
   131→              if (lesson) {
   132→                foundLesson = {
   133→                  ...lesson,
   134→                  Unit: { ...unit, Subject: subject }
   135→                };
   136→                
   137→                // Get related simulations
   138→                const simByLesson = getSimulationsByLessonId(lesson.id);
   139→                const simBySubject = getSimulationsBySubject(subject.nameAr);
   140→                const allSims = [...simByLesson];
   141→                simBySubject.forEach(sim => {
   142→                  if (!allSims.find(s => s.id === sim.id)) {
   143→                    allSims.push(sim);
   144→                  }
   145→                });
   146→                setRelatedSimulations(allSims);
   147→                
   148→                break;
   149→              }
   150→            }
   151→            if (foundLesson) break;
   152→          }
   153→          if (foundLesson) break;
   154→        }
   155→        
   156→        setLesson(foundLesson);
   157→      } catch (error) {
   158→        console.error("Error fetching lesson:", error);
   159→      } finally {
   160→        setLoading(false);
   161→      }
   162→    };
   163→    
   164→    fetchLesson();
   165→  }, [lessonId]);
   166→
   167→  const toggleTheme = () => {
   168→    setIsDark(!isDark);
   169→    document.documentElement.classList.toggle('dark');
   170→  };
   171→
   172→  const getTitle = () => {
   173→    if (!lesson) return "";
   174→    return language === "ar" ? lesson.titleAr : lesson.titleEn;
   175→  };
   176→
   177→  const getDescription = () => {
   178→    if (!lesson) return "";
   179→    return language === "ar" ? lesson.descriptionAr : lesson.descriptionEn;
   180→  };
   181→
   182→  const getIntroduction = () => {
   183→    if (!lesson) return "";
   184→    return language === "ar" ? lesson.introductionAr : lesson.introductionEn;
   185→  };
   186→
   187→  const getSummary = () => {
   188→    if (!lesson) return "";
   189→    return language === "ar" ? lesson.summaryAr : lesson.summaryEn;
   190→  };
   191→
   192→  const getObjectives = () => {
   193→    if (!lesson?.Objective) return [];
   194→    return lesson.Objective.sort((a, b) => a.order - b.order).map(obj => 
   195→      language === "ar" ? obj.textAr : obj.textEn
   196→    );
   197→  };
   198→
   199→  const getKeyConcepts = () => {
   200→    if (!lesson?.Concept) return [];
   201→    return lesson.Concept.sort((a, b) => a.order - b.order).map(concept => ({
   202→      term: language === "ar" ? concept.termAr : concept.termEn,
   203→      definition: language === "ar" ? concept.definitionAr : concept.definitionEn
   204→    }));
   205→  };
   206→
   207→  const getFormulas = () => {
   208→    if (!lesson?.Formula) return [];
   209→    return lesson.Formula.sort((a, b) => a.order - b.order).map(formula => ({
   210→      formula: formula.formula,
   211→      explanation: language === "ar" ? formula.explanationAr : formula.explanationEn
   212→    }));
   213→  };
   214→
   215→  const getExamples = () => {
   216→    if (!lesson?.Example) return [];

   218→      question: language === "ar" ? example.questionAr : example.questionEn,
   219→      solution: language === "ar" ? example.solutionAr : example.solutionEn,
   220→      steps: language === "ar" ? example.stepsAr?.split('\n') : example.stepsEn?.split('\n')
   221→    }));
   222→  };
   223→
   224→  const getQuestions = () => {
   225→    if (!lesson?.Question) return [];
   226→    return lesson.Question.sort((a, b) => a.order - b.order);
   227→  };
   228→
   229→  return (
   230→    <div className={`min-h-screen bg-gradient-to-br from-slate-50 via-purple-50 to-pink-50 dark:from-slate-950 dark:via-purple-950 dark:to-pink-950 ${isRTL ? "rtl" : "ltr"}`} dir={isRTL ? "rtl" : "ltr"}>
   231→      {/* Header */}
   232→      <header className="sticky top-0 z-50 bg-white/80 dark:bg-slate-900/80 backdrop-blur-lg border-b border-slate-200 dark:border-slate-800">
   233→        <div className="max-w-7xl mx-auto px-4 py-4">
   234→          <div className="flex items-center justify-between">
   235→            <div className="flex items-center gap-4">
   236→              <Link href="/platform" className="flex items-center gap-2 hover:opacity-80 transition">
   237→                <img src="/logo.jpeg" alt="SmartEdu" className="w-10 h-10 rounded-xl object-cover" />
   238→                <span className="font-bold text-xl bg-gradient-to-r from-purple-600 to-pink-600 bg-clip-text text-transparent hidden sm:inline">
   239→                  {t("تعلم ذكي", "SmartEdu")}
   240→                </span>
   241→              </Link>
   242→            </div>
   243→
   244→            <div className="flex items-center gap-2">
   245→              {lesson?.Unit?.Subject && (
   246→                <Link href={`/platform/subject/${lesson.Unit.Subject.id}`}>
   247→                  <Button variant="ghost" size="sm" className="gap-2">
   248→                    <ArrowLeft className="w-4 h-4" />
   249→                    {t("العودة", "Back")}
   250→                  </Button>
   251→                </Link>
   252→              )}
   253→              <Button
   254→                variant="ghost"
   255→                size="sm"
   256→                onClick={toggleLanguage}
   257→                className="gap-2"
   258→              >
   259→                <Globe className="w-4 h-4" />
   260→                {language === "ar" ? "EN" : "عربي"}
   261→              </Button>
   262→              <Button
   263→                variant="ghost"
   264→                size="icon"
   265→                onClick={toggleTheme}
   266→              >
   267→                {isDark ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
   268→              </Button>
   269→            </div>
   270→          </div>
   271→        </div>
   272→      </header>
   273→
   274→      {/* Main Content */}
   275→      <main className="max-w-5xl mx-auto px-4 py-8">
   276→        {loading ? (
   277→          <div className="flex items-center justify-center py-20">
   278→            <div className="animate-spin w-12 h-12 border-4 border-purple-500 border-t-transparent rounded-full"></div>
   279→          </div>
   280→        ) : !lesson ? (
   281→          <div className="text-center py-20">
   282→            <h2 className="text-2xl font-bold text-slate-800 dark:text-white mb-4">
   283→              {t("الدرس غير موجود", "Lesson not found")}
   284→            </h2>
   285→            <Link href="/platform">
   286→              <Button>
   287→                {t("العودة للمنصة", "Back to Platform")}
   288→              </Button>
   289→            </Link>
   290→          </div>
   291→        ) : (
   292→          <>
   293→            {/* Breadcrumb */}
   294→            <nav className="flex items-center gap-2 text-sm text-slate-500 mb-6 flex-wrap">
   295→              <Link href="/platform" className="hover:text-purple-600 transition">
   296→                {t("المنصة", "Platform")}
   297→              </Link>
   298→              <ChevronRight className="w-4 h-4" />
   299→              {lesson.Unit?.Subject && (
   300→                <>
   301→                  <Link 
   302→                    href={`/platform/subject/${lesson.Unit.Subject.id}`} 
   303→                    className="hover:text-purple-600 transition"
   304→                  >
   305→                    {language === "ar" ? lesson.Unit.Subject.nameAr : lesson.Unit.Subject.nameEn}
   306→                  </Link>
   307→                  <ChevronRight className="w-4 h-4" />
   308→                </>
   309→              )}
   310→              <span className="text-slate-800 dark:text-white font-medium line-clamp-1">
   311→                {getTitle()}
   312→              </span>
   313→            </nav>
   314→
   315→            {/* Lesson Header */}
   316→            <motion.div
   317→              initial={{ opacity: 0, y: 20 }}
   318→              animate={{ opacity: 1, y: 0 }}
   319→              className="mb-8"
   320→            >
   321→              <div className="flex items-start gap-4 mb-4">
   322→                <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${
   323→                  lesson.isFree 
   324→                    ? "bg-green-100 dark:bg-green-900/30" 
   325→                    : "bg-purple-100 dark:bg-purple-900/30"
   326→                }`}>
   327→                  {lesson.isFree ? (
   328→                    <Play className="w-6 h-6 text-green-600" />
   329→                  ) : (
   330→                    <Lock className="w-6 h-6 text-purple-600" />
   331→                  )}
   332→                </div>
   333→                <div className="flex-1">
   334→                  <h1 className="text-2xl md:text-3xl font-bold text-slate-800 dark:text-white mb-2">
   335→                    {getTitle()}
   336→                  </h1>
   337→                  <p className="text-slate-600 dark:text-slate-400 mb-3">
   338→                    {getDescription()}
   339→                  </p>
   340→                  <div className="flex items-center gap-4 text-sm text-slate-500">
   341→                    <div className="flex items-center gap-1">
   342→                      <Clock className="w-4 h-4" />
   343→                      {lesson.duration} {t("دقيقة", "min")}
   344→                    </div>
   345→                    {lesson.isFree && (
   346→                      <Badge className="bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400 border-0">
   347→                        {t("مجاني", "Free")}
   348→                      </Badge>
   349→                    )}
   350→                    {relatedSimulations.length > 0 && (
   351→                      <Badge className="bg-blue-100 text-blue-700 dark:bg-blue-900/30 dark:text-blue-400 border-0">
   352→                        <FlaskConical className="w-3 h-3 mr-1" />
   353→                        {relatedSimulations.length} {t("محاكاة", "simulations")}
   354→                      </Badge>
   355→                    )}
   356→                  </div>
   357→                </div>
   358→              </div>
   359→            </motion.div>
   360→
   361→            {/* Lesson Content Tabs */}
   362→            <Tabs defaultValue="content" className="space-y-6">
   363→              <TabsList className="grid grid-cols-2 md:grid-cols-6 gap-2 h-auto p-2 bg-white dark:bg-slate-800 rounded-xl">
   364→                <TabsTrigger value="content" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   365→                  <BookOpen className="w-4 h-4 mr-2" />
   366→                  {t("المحتوى", "Content")}
   367→                </TabsTrigger>
   368→                <TabsTrigger value="concepts" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   369→                  <Lightbulb className="w-4 h-4 mr-2" />
   370→                  {t("المفاهيم", "Concepts")}
   371→                </TabsTrigger>
   372→                <TabsTrigger value="formulas" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   373→                  <Target className="w-4 h-4 mr-2" />
   374→                  {t("القوانين", "Formulas")}
   375→                </TabsTrigger>
   376→                <TabsTrigger value="examples" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   377→                  <FileText className="w-4 h-4 mr-2" />
   378→                  {t("أمثلة", "Examples")}
   379→                </TabsTrigger>
   380→                <TabsTrigger value="simulations" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   381→                  <Beaker className="w-4 h-4 mr-2" />
   382→                  {t("محاكيات", "Simulations")}
   383→                </TabsTrigger>
   384→                <TabsTrigger value="quiz" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   385→                  <HelpCircle className="w-4 h-4 mr-2" />
   386→                  {t("اختبار", "Quiz")}
   387→                </TabsTrigger>
   388→              </TabsList>
   389→
   390→              {/* Content Tab */}
   391→              <TabsContent value="content">
   392→                <motion.div
   393→                  initial={{ opacity: 0, y: 10 }}
   394→                  animate={{ opacity: 1, y: 0 }}
   395→                  className="space-y-6"
   396→                >
   397→                  {getObjectives().length > 0 && (
   398→                    <Card className="border-0 shadow-md">
   399→                      <CardContent className="p-6">
   400→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   401→                          <Target className="w-5 h-5 text-purple-500" />
   402→                          {t("أهداف الدرس", "Lesson Objectives")}
   403→                        </h3>
   404→                        <ul className="space-y-2">
   405→                          {getObjectives().map((obj: string, i: number) => (
   406→                            <li key={i} className="flex items-start gap-3">
   407→                              <CheckCircle className="w-5 h-5 text-green-500 mt-0.5 flex-shrink-0" />
   408→                              <span className="text-slate-600 dark:text-slate-300">{obj}</span>
   409→                            </li>
   410→                          ))}
   411→                        </ul>
   412→                      </CardContent>
   413→                    </Card>
   414→                  )}
   415→
   416→                  {getIntroduction() && (
   417→                    <Card className="border-0 shadow-md">
   418→                      <CardContent className="p-6">
   419→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   420→                          <BookOpen className="w-5 h-5 text-purple-500" />
   421→                          {t("مقدمة الدرس", "Introduction")}
   422→                        </h3>
   423→                        <div className="prose dark:prose-invert max-w-none text-slate-600 dark:text-slate-300 whitespace-pre-wrap">
   424→                          {getIntroduction()}
   425→                        </div>
   426→                      </CardContent>
   427→                    </Card>
   428→                  )}
   429→
   430→                  {getSummary() && (
   431→                    <Card className="border-0 shadow-md bg-gradient-to-r from-purple-50 to-pink-50 dark:from-purple-900/20 dark:to-pink-900/20">
   432→                      <CardContent className="p-6">
   433→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   434→                          <FileText className="w-5 h-5 text-purple-500" />
   435→                          {t("ملخص الدرس", "Summary")}
   436→                        </h3>
   437→                        <div className="prose dark:prose-invert max-w-none text-slate-600 dark:text-slate-300 whitespace-pre-wrap">
   438→                          {getSummary()}
   439→                        </div>
   440→                      </CardContent>
   441→                    </Card>
   442→                  )}
   443→                </motion.div>
   444→              </TabsContent>
   445→
   446→              {/* Concepts Tab */}
   447→              <TabsContent value="concepts">
   448→                <motion.div
   449→                  initial={{ opacity: 0, y: 10 }}
   450→                  animate={{ opacity: 1, y: 0 }}
   451→                >
   452→                  <Card className="border-0 shadow-md">
   453→                    <CardContent className="p-6">
   454→                      <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-6 flex items-center gap-2">
   455→                        <Lightbulb className="w-5 h-5 text-purple-500" />
   456→                        {t("المفاهيم الأساسية", "Key Concepts")}
   457→                      </h3>
   458→                      {getKeyConcepts().length === 0 ? (
   459→                        <p className="text-slate-500 text-center py-8">

   461→                        </p>
   462→                      ) : (
   463→                        <div className="space-y-4">
   464→                          {getKeyConcepts().map((concept: any, i: number) => (
   465→                            <div key={i} className="p-4 bg-slate-50 dark:bg-slate-800 rounded-xl">
   466→                              <h4 className="font-bold text-purple-600 dark:text-purple-400 mb-2">
   467→                                {concept.term}
   468→                              </h4>
   469→                              <p className="text-slate-600 dark:text-slate-300">
   470→                                {concept.definition}
   471→                              </p>
   472→                            </div>
   473→                          ))}
   474→                        </div>
   475→                      )}
   476→                    </CardContent>
   477→                  </Card>
   478→                </motion.div>
   479→              </TabsContent>
   480→
   481→              {/* Formulas Tab */}
   482→              <TabsContent value="formulas">
   483→                <motion.div
   484→                  initial={{ opacity: 0, y: 10 }}
   485→                  animate={{ opacity: 1, y: 0 }}
   486→                >
   487→                  <Card className="border-0 shadow-md">
   488→                    <CardContent className="p-6">
   489→                      <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-6 flex items-center gap-2">

   502→                                {formula.formula}
   503→                              </div>
   504→                              <p className="text-slate-600 dark:text-slate-300 text-center">
   505→                                {formula.explanation}

   507→                            </div>
   508→                          ))}
   509→                        </div>
   510→                      )}
   511→                    </CardContent>
   512→                  </Card>
   513→                </motion.div>
   514→              </TabsContent>
   515→
   516→              {/* Examples Tab */}
   517→              <TabsContent value="examples">
   518→                <motion.div
   519→                  initial={{ opacity: 0, y: 10 }}
   520→                  animate={{ opacity: 1, y: 0 }}
   521→                >
   522→                  <Card className="border-0 shadow-md">
   523→                    <CardContent className="p-6">
