   112→   112→  
   113→   113→

   239→   239→              </Link>

   246→   246→                    <ArrowLeft className="w-4 h-4" />

   252→   252→                variant="ghost"
   253→   253→                size="sm"

   279→   279→          <div className="text-center py-20">
   280→   280→            <h2 className="text-2xl font-bold text-slate-800 dark:text-white mb-4">
   281→   281→              {t("الدرس غير موجود", "Lesson not found")}
   282→   282→            </h2>
   283→   283→            <Link href="/platform">
   284→   284→              <Button>
   285→   285→                {t("العودة للمنصة", "Back to Platform")}
   286→   286→              </Button>
   287→   287→            </Link>
   288→   288→          </div>
   289→   289→        ) : (
   290→   290→          <>
   291→   291→            {/* Breadcrumb */}
   292→   292→            <nav className="flex items-center gap-2 text-sm text-slate-500 mb-6 flex-wrap">
   293→   293→              <Link href="/platform" className="hover:text-purple-600 transition">
   294→   294→                {t("المنصة", "Platform")}
   295→   295→              </Link>

   300→   300→                    href={`/platform/subject/${lesson.Unit.Subject.id}`} 
   301→   301→                    className="hover:text-purple-600 transition"
   302→   302→                  >
   303→   303→                    {language === "ar" ? lesson.Unit.Subject.nameAr : lesson.Unit.Subject.nameEn}
   304→   304→                  </Link>

   313→   313→            {/* Lesson Header */}

   319→   319→              <div className="flex items-start gap-4 mb-4">
   320→   320→                <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${
   321→   321→                  lesson.isFree

   328→   328→                    <Lock className="w-6 h-6 text-purple-600" />
