   591→        return;

   593→      
   594→      for (const lessonData of lessons) {

   625→

   627→          for (let i = 0; i < lessonData.concepts.ar.length; i++) {

   629→              data: {
   630→                lessonId: lesson.id,
   631→                termAr: lessonData.concepts.ar[i].term,

   642→            await db.formula.create({
   643→              data: {

   652→          
   653→          // إضافة الأمثلة
   654→          for (let i = 0; i < lessonData.examples.ar.length; i++) {
   655→            await db.example.create({
   656→              data: {
   657→                lessonId: lesson.id,
   658→                questionAr: lessonData.examples.ar[i].question,
   659→                questionEn: lessonData.examples.en[i]?.question || "",
   660→                solutionAr: lessonData.examples.ar[i].solution,
   661→                solutionEn: lessonData.examples.en[i]?.solution || "",

   673→              await db.lessonSimulator.create({
   674→                data: {
   675→                  lessonId: lesson.id,
   676→                  simulatorId: simulator.id,
   677→                },
   678→              });
   679→            }
   680→          }
   681→          
   682→          // إضافة الأسئلة
   683→          for (const q of lessonData.questions) {
   684→            await db.question.create({
   685→              data: {
   686→                lessonId: lesson.id,
   687→                type: q.type,
   688→                questionAr: q.questionAr,
   689→                questionEn: q.questionEn,
   690→                optionsAr: JSON.stringify(q.optionsAr),
   691→                optionsEn: JSON.stringify(q.optionsEn),
   692→                answer: q.answer,
   693→                points: q.points,
