   354→function CircleDiagram({
   355→  data,
   356→  language,
   357→}: {
   358→  data: CircleData;
   359→  language: "ar" | "en";
   360→}) {
   361→  const dir = language === "ar" ? "rtl" : "ltr";
   362→  const isRTL = language === "ar";
   363→
   364→  // حساب الزوايا والأقواس
   365→  const total = data.segments.reduce((sum, seg) => sum + seg.value, 0);
   366→  
   367→  const segmentsWithAngles = useMemo(() => {
   368→    // استخدام reduce لبناء المصفوفة مع تتبع الزاوية الحالية
   369→    const result = data.segments.reduce<{
   370→      segments: Array<{
   371→        label: string;
   372→        labelAr: string;
   373→        value: number;
   374→        color: string;
   375→        description?: string;
   376→        descriptionAr?: string;
   377→        percentage: number;
   378→        startAngle: number;
   379→        endAngle: number;
   380→      }>;
   381→      currentAngle: number;
   382→    }>((acc, seg, index) => {
   383→      const percentage = (seg.value / total) * 100;
   384→      const angle = (seg.value / total) * 360;
   385→      const startAngle = acc.currentAngle;
   386→      const endAngle = acc.currentAngle + angle;
   387→
   388→      acc.segments.push({
   389→        ...seg,
   390→        percentage,
   391→        startAngle,
   392→        endAngle,
   393→        color: seg.color || SEGMENT_COLORS[index % SEGMENT_COLORS.length],
   394→      });
   395→
   396→      acc.currentAngle = endAngle;
   397→      return acc;
   398→    }, { segments: [], currentAngle: -90 });
   399→
   400→    return result.segments;
   401→  }, [data.segments, total]);
   402→
   403→  // إنشاء مسار القوس SVG
   404→  const createArcPath = (startAngle: number, endAngle: number, radius: number) => {
   405→    const startRad = (startAngle * Math.PI) / 180;
   406→    const endRad = (endAngle * Math.PI) / 180;
   407→    
   408→    const startX = 100 + radius * Math.cos(startRad);
   409→    const startY = 100 + radius * Math.sin(startRad);
   410→    const endX = 100 + radius * Math.cos(endRad);
   411→    const endY = 100 + radius * Math.sin(endRad);
   412→    
   413→    const largeArc = endAngle - startAngle > 180 ? 1 : 0;
   414→    
   415→    return `M 100 100 L ${startX} ${startY} A ${radius} ${radius} 0 ${largeArc} 1 ${endX} ${endY} Z`;
   416→  };
   417→
   418→  return (
   419→    <motion.div
   420→      className="space-y-6"
   421→      variants={containerVariants}
   422→      initial="hidden"
   423→      animate="visible"
   424→      dir={dir}
   425→    >
   426→      <div className="flex flex-col lg:flex-row items-center gap-8">
   427→        {/* الدائرة */}
   428→        <motion.div
   429→          variants={scaleVariants}
   430→          className="relative w-64 h-64 lg:w-80 lg:h-80"
   431→        >
   432→          <svg viewBox="0 0 200 200" className="w-full h-full transform -rotate-0">
   433→            {/* خلفية الدائرة */}
   434→            <circle
   435→              cx="100"
   436→              cy="100"
   437→              r="80"
   438→              fill="none"
   439→              stroke="currentColor"
   440→              strokeWidth="2"
   441→              className="text-slate-200 dark:text-slate-700"
   442→            />
   443→            
   444→            {/* الأقسام */}
   445→            {segmentsWithAngles.map((seg, index) => (
   446→              <motion.path
   447→                key={index}
   448→                d={createArcPath(seg.startAngle, seg.endAngle, 80)}
   449→                fill={seg.color}
   450→                initial={{ opacity: 0, scale: 0.8 }}
   451→                animate={{ opacity: 1, scale: 1 }}
   452→                transition={{ duration: 0.5, delay: index * 0.1 }}
   453→                className="hover:opacity-80 transition-opacity cursor-pointer"
   454→              />
   455→            ))}
   456→            
   457→            {/* الدائرة المركزية */}
   458→            <circle
   459→              cx="100"
   460→              cy="100"
   461→              r="40"
   462→              fill="white"
   463→              className="dark:fill-slate-800"
   464→            />
   465→            
   466→            {/* النص المركزي */}
   467→            <text
   468→              x="100"
   469→              y="100"
   470→              textAnchor="middle"
   471→              dominantBaseline="middle"
   472→              className="text-sm font-bold fill-slate-700 dark:fill-slate-200"
   473→            >
   474→              {isRTL ? data.centerLabelAr : data.centerLabel || (data.showPercentage !== false ? `${total}%` : total)}
   475→            </text>
   476→          </svg>
   477→        </motion.div>
   478→
   479→        {/* التسميات */}
   480→        <div className="flex-1 space-y-3">
   481→          {segmentsWithAngles.map((seg, index) => (
   482→            <motion.div
   483→              key={index}
   484→              variants={itemVariants}
   485→              className="flex items-center gap-3 p-3 bg-slate-50 dark:bg-slate-800 rounded-lg hover:shadow-md transition-shadow"
   486→            >
   487→              <div
   488→                className="w-4 h-4 rounded-full shrink-0"
   489→                style={{ backgroundColor: seg.color }}
   490→              />
   491→              <div className="flex-1">
   492→                <div className="flex items-center justify-between">
   493→                  <span className="font-medium">{isRTL ? seg.labelAr : seg.label}</span>
   494→                  <Badge variant="outline" className="ml-2">
   495→                    {data.showPercentage !== false ? `${seg.percentage.toFixed(1)}%` : seg.value}
   496→                  </Badge>
   497→                </div>
   498→                {(isRTL ? seg.descriptionAr : seg.description) && (
   499→                  <p className="text-sm text-slate-500 mt-1">
   500→                    {isRTL ? seg.descriptionAr : seg.description}
   501→                  </p>
   502→                )}
   503→              </div>
   504→            </motion.div>
   505→          ))}
   506→        </div>
   507→      </div>
   508→    </motion.div>
   509→  );
   510→}

   584→                    <span className="text-purple-500">{iconMap[event.icon]}</span>
   585→                  )}
   586→                  {isRTL ? event.titleAr : event.title}
   587→                </h3>
   588→
   589→                {/* الوصف */}
   590→                <p className="text-slate-600 dark:text-slate-300">
   591→                  {isRTL ? event.descriptionAr : event.description}
   592→                </p>
   593→              </CardContent>
   594→            </Card>
   595→          </motion.div>
   596→        ))}
   597→      </div>
   598→    </motion.div>
   599→  );

   604→// ==========================================
   605→export function Infographic({
   606→  type,
   607→  data,
   608→  language,
   609→  title,
   610→  titleAr,
   611→  className = "",
   612→}: InfographicProps) {
   613→  const dir = language === "ar" ? "rtl" : "ltr";
   614→  const isRTL = language === "ar";
   615→
   616→  // أيقونة حسب النوع
   617→  const typeIcons = {
   618→    comparison: <TrendingUp className="w-5 h-5" />,
   619→    process: <GitBranch className="w-5 h-5" />,
   620→    circle: <Circle className="w-5 h-5" />,
   621→    timeline: <Clock className="w-5 h-5" />,
   622→  };
   623→
   624→  // ألوان حسب النوع
   625→  const typeColors = {
   626→    comparison: "from-emerald-500 to-teal-500",
   627→    process: "from-blue-500 to-indigo-500",
   628→    circle: "from-purple-500 to-pink-500",
   629→    timeline: "from-amber-500 to-orange-500",
   630→  };
   631→
   632→  // عنوان افتراضي حسب النوع
   633→  const defaultTitles = {
   634→    comparison: { ar: "مخطط المقارنة", en: "Comparison Chart" },
   635→    process: { ar: "خطوات العملية", en: "Process Flow" },
   636→    circle: { ar: "التوزيع النسبي", en: "Distribution Chart" },
   637→    timeline: { ar: "الجدول الزمني", en: "Timeline" },
   638→  };
   639→
   640→  const displayTitle = (isRTL ? titleAr : title) || defaultTitles[type][language];
   641→
   642→  // عرض المخطط المناسب
   643→  const renderChart = () => {
   644→    switch (type) {
   645→      case "comparison":
   646→        return <ComparisonChart data={data as ComparisonData} language={language} />;
   647→      case "process":
   648→        return <ProcessFlow data={data as ProcessData} language={language} />;
   649→      case "circle":
   650→        return <CircleDiagram data={data as CircleData} language={language} />;
   651→      case "timeline":
   652→        return <Timeline data={data as TimelineData} language={language} />;
   653→      default:
   654→        return null;
   655→    }
   656→  };
   657→
   658→  return (
   659→    <motion.div
   660→      initial={{ opacity: 0, y: 20 }}
   661→      animate={{ opacity: 1, y: 0 }}
   662→      transition={{ duration: 0.5 }}
   663→      className={className}
   664→      dir={dir}
   665→    >
   666→      <Card className="border-0 shadow-lg overflow-hidden">
   667→        {/* شريط العنوان الملون */}
   668→        <div className={`h-2 bg-gradient-to-r ${typeColors[type]}`} />
   669→        
   670→        <CardHeader className="pb-2">
   671→          <CardTitle className="flex items-center gap-2">
   672→            <div className={`w-10 h-10 rounded-lg flex items-center justify-center ${
   673→              type === "comparison" ? "bg-emerald-100 text-emerald-600" :
   674→              type === "process" ? "bg-blue-100 text-blue-600" :
   675→              type === "circle" ? "bg-purple-100 text-purple-600" :
   676→              "bg-amber-100 text-amber-600"
   677→            }`}>
   678→              {typeIcons[type]}
   679→            </div>
   680→            {displayTitle}
   681→          </CardTitle>
   682→        </CardHeader>
   683→        
   684→        <CardContent className="pt-4">
   685→          {renderChart()}
   686→        </CardContent>
   687→      </Card>
   688→    </motion.div>
   689→  );
   690→}
