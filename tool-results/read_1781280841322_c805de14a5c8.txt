   179→                  </div>

   181→                </motion.div>

   183→            </div>
   184→          )}
   185→        </div>
   186→      )}

   520→                <div>
   521→                  <Button
   522→                    variant="ghost"
   523→                    onClick={() => {
   524→                      if (selectedSpecialization) {
   525→                        setSelectedSpecialization(null);
   526→                      } else {
   527→                        setSelectedYear(null);
   528→                      }
   529→                    }}
   530→                    className="mb-4"
   531→                  >
   532→                    {language === "ar" ? <ChevronRight className="w-4 h-4" /> : <ChevronLeft className="w-4 h-4" />}
   533→                    {language === "ar" ? "العودة" : "Back"}
   534→                  </Button>
   535→
   536→                  <div className="text-center mb-6">
   537→                    <h2 className="text-2xl font-bold text-slate-800 dark:text-white mb-2">
   538→                      {language === "ar" 
   539→                        ? `${academicYears.find(y => y.code === selectedYear)?.nameAr}${selectedSpecialization ? ` - ${specializations.find(s => s.code === selectedSpecialization)?.nameAr}` : ''}`
   540→                        : `${academicYears.find(y => y.code === selectedYear)?.nameEn}${selectedSpecialization ? ` - ${specializations.find(s => s.code === selectedSpecialization)?.nameEn}` : ''}`}
   541→                    </h2>
   542→                  </div>
   543→
   544→                  <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   545→                    {getSubjectsForYearAndSpec().map((subject, index) => {
   546→                      const Icon = subjectIcons[subject.icon] || BookOpen;
   547→                      
   548→                      return (
   549→                        <motion.div
   550→                          key={subject.id}
   551→                          initial={{ opacity: 0, y: 20 }}
   552→                          animate={{ opacity: 1, y: 0 }}
   553→                          transition={{ delay: index * 0.05 }}
   554→                        >
   555→                          <Card
   556→                            onClick={() => setSelectedSubject(subject)}
   557→                            className="cursor-pointer hover:border-emerald-500/50 transition-all"
   558→                          >
   559→                            <CardContent className="p-4">
   560→                              <div className="flex items-center gap-3">
   561→                                <div className={`p-3 rounded-xl bg-gradient-to-br ${
   562→                                  subject.color === 'emerald' ? 'from-emerald-400 to-teal-500' :
   563→                                  subject.color === 'purple' ? 'from-purple-400 to-pink-500' :
   564→                                  subject.color === 'blue' ? 'from-blue-400 to-cyan-500' :
   565→                                  subject.color === 'amber' ? 'from-amber-400 to-orange-500' :
   566→                                  subject.color === 'rose' ? 'from-rose-400 to-pink-500' :
   567→                                  subject.color === 'teal' ? 'from-teal-400 to-cyan-500' :
   568→                                  'from-slate-400 to-slate-500'
   569→                                }`}>
   570→                                  <Icon className="w-6 h-6 text-white" />
   571→                                </div>
   572→                                <div>
   573→                                  <h3 className="font-bold text-slate-800 dark:text-white">
   574→                                    {language === "ar" ? subject.nameAr : subject.nameEn}
   575→                                  </h3>
   576→                                  <p className="text-sm text-slate-500">
   577→                                    {subject.units?.length || 0} {language === "ar" ? "وحدة" : "units"}
   578→                                  </p>
   579→                                </div>
   580→                              </div>
   581→                            </CardContent>
   582→                          </Card>
   583→                        </motion.div>
   584→                      );
   585→                    })}
   586→                  </div>
   587→                </div>
   588→              )}
   589→
   590→              {/* عرض الوحدات والدروس */}
   591→              {selectedSubject && (
   592→                <div>
   593→                  <Button
   594→                    variant="ghost"
   595→                    onClick={() => setSelectedSubject(null)}
   596→                    className="mb-4"
   597→                  >
   598→                    {language === "ar" ? <ChevronRight className="w-4 h-4" /> : <ChevronLeft className="w-4 h-4" />}
   599→                    {language === "ar" ? "العودة للمواد" : "Back to Subjects"}
   600→                  </Button>
   601→
   602→                  <div className="mb-6">
   603→                    <h2 className="text-2xl font-bold text-slate-800 dark:text-white">
   604→                      {language === "ar" ? selectedSubject.nameAr : selectedSubject.nameEn}
   605→                    </h2>
   606→                  </div>
   607→
   608→                  {/* الوحدات */}
   609→                  {selectedSubject.units?.map((unit, unitIndex) => (
   610→                    <UnitCard 
   611→                      key={unit.id} 
   612→                      unit={unit} 
   613→                      language={language}
   614→                      onSelectLesson={handleSelectLesson}
   615→                    />
   616→                  ))}
   617→                </div>
   618→              )}
   619→
   620→              {/* إحصائيات */}
   621→              {!selectedYear && (
   622→                <motion.div
   623→                  initial={{ opacity: 0, y: 20 }}
   624→                  animate={{ opacity: 1, y: 0 }}
   625→                  transition={{ delay: 0.5 }}
   626→                  className="mt-8 grid grid-cols-2 md:grid-cols-4 gap-4"
   627→                >
   628→                  <Card className="text-center p-4">
   629→                    <Database className="w-8 h-8 mx-auto mb-2 text-emerald-500" />
   630→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">3</div>
   631→                    <div className="text-sm text-slate-500">{language === "ar" ? "سنوات دراسية" : "Years"}</div>
   632→                  </Card>
   633→                  <Card className="text-center p-4">
   634→                    <BookOpen className="w-8 h-8 mx-auto mb-2 text-blue-500" />
   635→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">12</div>
   636→                    <div className="text-sm text-slate-500">{language === "ar" ? "محاكي تفاعلي" : "Simulators"}</div>
   637→                  </Card>
   638→                  <Card className="text-center p-4">
   639→                    <Users className="w-8 h-8 mx-auto mb-2 text-purple-500" />
   640→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">3</div>
   641→                    <div className="text-sm text-slate-500">{language === "ar" ? "تخصصات" : "Tracks"}</div>
   642→                  </Card>
   643→                  <Card className="text-center p-4">
   644→                    <Brain className="w-8 h-8 mx-auto mb-2 text-amber-500" />
   645→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">{points}</div>
   646→                    <div className="text-sm text-slate-500">{language === "ar" ? "نقطة" : "Points"}</div>
   647→                  </Card>
   648→                </motion.div>
   649→              )}
   650→            </motion.div>
   651→          )}
   652→        </AnimatePresence>
   653→      </main>
   654→
   655→      {/* الفوتر */}
   656→      <footer className="mt-auto border-t border-slate-200 dark:border-slate-800 bg-white/80 dark:bg-slate-900/80 backdrop-blur-sm">
   657→        <div className="container mx-auto px-4 py-4">
   658→          <div className="flex flex-col md:flex-row items-center justify-between gap-4">
   659→            <p className="text-sm text-slate-500 dark:text-slate-400 text-center md:text-start">
   660→              {language === "ar" 
   661→                ? "© 2024 ثانوية تفاعلية - المنهج المصري الكامل" 
   662→                : "© 2024 Interactive HS - Complete Egyptian Curriculum"}
   663→            </p>
   664→            <div className="flex items-center gap-4">
   665→              <Badge variant="secondary" className="gap-1">
   666→                <Database className="w-3 h-3" />
   667→                API Ready
   668→              </Badge>
   669→              <Badge variant="secondary" className="gap-1">
   670→                <Sparkles className="w-3 h-3" />
   671→                v3.0
   672→              </Badge>
   673→            </div>
   674→          </div>
   675→        </div>
   676→      </footer>
   677→
   678→      {/* نافذة المكافآت */}
   679→      <RewardsSystem
   680→        isOpen={showRewards}
   681→        onClose={() => setShowRewards(false)}
   682→        userProgress={userProgress}
   683→        language={language}
   684→      />
   685→    </div>
   686→  );
   687→}
