    87→function UnitCard({ 
    88→  unit, 
    89→  language,

    96→  const [lessons, setLessons] = useState<any[]>([]);

    98→  const [expanded, setExpanded] = useState(false);
    99→  const hasLoaded = useRef(false);
   100→
   101→  useEffect(() => {
   102→    if (!expanded || hasLoaded.current) return;
   103→    
   104→    hasLoaded.current = true;
   105→    
   106→    const loadLessons = async () => {
   107→      setLoading(true);
   108→      try {
   109→        const res = await fetch(`/api/units/${unit.id}/lessons`);
   110→        const data = await res.json();

   123→    <Card className="mb-4 overflow-hidden">

   126→        onClick={() => setExpanded(!expanded)}

   128→        <h3 className="font-bold text-slate-800 dark:text-white">
   129→          {language === "ar" ? unit.nameAr : unit.nameEn}
   130→        </h3>

   136→        </div>
   137→      </div>

   139→      {expanded && (
   140→        <div className="p-4 border-t border-slate-200 dark:border-slate-700">
   141→          {loading ? (
   142→            <div className="text-center py-4">

   144→            </div>

   147→              <Database className="w-8 h-8 mx-auto mb-2 opacity-50" />

   149→                {language === "ar" ? "لا توجد دروس بعد" : "No lessons yet"}
   150→              </p>
   151→            </div>
   152→          ) : (
   153→            <div className="space-y-2">
   154→              {lessons.map((lesson) => (
   155→                <motion.div
   156→                  key={lesson.id}
   157→                  initial={{ opacity: 0, x: -10 }}
   158→                  animate={{ opacity: 1, x: 0 }}
   159→                  className="flex items-center justify-between p-3 rounded-lg bg-slate-50 dark:bg-slate-800/50 hover:bg-slate-100 dark:hover:bg-slate-800 cursor-pointer transition-colors"
   160→                  onClick={() => onSelectLesson(lesson.id)}
   161→                >
   162→                  <div className="flex items-center gap-3">
   163→                    <div className="w-8 h-8 rounded-full bg-purple-100 dark:bg-purple-900/30 flex items-center justify-center">
   164→                      {lesson.isFree ? (
   165→                        <Play className="w-4 h-4 text-purple-600 dark:text-purple-400" />
   166→                      ) : (
   167→                        <Lock className="w-4 h-4 text-slate-400" />
   168→                      )}
   169→                    </div>
   170→                    <div>
   171→                      <h4 className="font-medium text-slate-800 dark:text-white">
   172→                        {language === "ar" ? lesson.titleAr : lesson.titleEn}
   173→                      </h4>
   174→                      <div className="flex items-center gap-2 text-xs text-slate-500">
   175→                        <Clock className="w-3 h-3" />
   176→                        <span>{lesson.duration} {language === "ar" ? "دقيقة" : "min"}</span>
   177→                        {lesson.simulators?.length > 0 && (
   178→                          <>
   179→                            <span>•</span>
   180→                            <Zap className="w-3 h-3 text-amber-500" />
   181→                            <span>{lesson.simulators.length} {language === "ar" ? "محاكي" : "sims"}</span>
   182→                          </>
   183→                        )}
   184→                      </div>
   185→                    </div>
   186→                  </div>
   187→                  <ChevronLeft className="w-4 h-4 text-slate-400" />
   188→                </motion.div>
   189→              ))}
   190→            </div>
   191→          )}
   192→        </div>
   193→      )}
   194→    </Card>
   195→  );
   196→}
