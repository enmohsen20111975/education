    87→function UnitCard({

    98→  const [expanded, setExpanded] = useState(false);

   118→        className="p-4 bg-gradient-to-r from-slate-100 to-slate-50 dark:from-slate-800 dark:to-slate-900 cursor-pointer flex items-center justify-between"

   121→        <h3 className="font-bold text-slate-800 dark:text-white">
   122→          {language === "ar" ? unit.nameAr : unit.nameEn}
   123→        </h3>
   124→        <div className="flex items-center gap-2">
   125→          <Badge variant="secondary" className="text-xs">
   126→            {lessons.length || "..."} {language === "ar" ? "درس" : "lessons"}

   128→          <ChevronLeft className={`w-4 h-4 transition-transform ${expanded ? "-rotate-90" : ""}`} />
   129→        </div>
   130→      </div>

   132→      {expanded && (
   133→        <div className="p-4 border-t border-slate-200 dark:border-slate-700">
   134→          {loading ? (
   135→            <div className="text-center py-4">

   140→              <Database className="w-8 h-8 mx-auto mb-2 opacity-50" />

   143→              </p>
   144→            </div>

   146→            <div className="space-y-2">
   147→              {lessons.map((lesson) => (
   148→                <motion.div
   149→                  key={lesson.id}
   150→                  initial={{ opacity: 0, x: -10 }}

   156→                    <div className="w-8 h-8 rounded-full bg-emerald-100 dark:bg-emerald-900/30 flex items-center justify-center">

   158→                        <Play className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />

   160→                        <Lock className="w-4 h-4 text-slate-400" />
   161→                      )}
   162→                    </div>
   163→                    <div>
   164→                      <h4 className="font-medium text-slate-800 dark:text-white">
   165→                        {language === "ar" ? lesson.titleAr : lesson.titleEn}
   166→                      </h4>
   167→                      <div className="flex items-center gap-2 text-xs text-slate-500">
   168→                        <Clock className="w-3 h-3" />

   170→                        {lesson.simulators?.length > 0 && (
   171→                          <>
   172→                            <span>•</span>

   176→                        )}
   177→                      </div>
   178→                    </div>
   179→                  </div>
   180→                  <ChevronLeft className="w-4 h-4 text-slate-400" />
   181→                </motion.div>
   182→              ))}
   183→            </div>
   184→          )}
   185→        </div>
   186→      )}
   187→    </Card>
   188→  );
   189→}

   529→                  >

   531→                    {language === "ar" ? "العودة" : "Back"}
   532→                  </Button>
   533→
   534→                  <div className="text-center mb-6">
   535→                    <h2 className="text-2xl font-bold text-slate-800 dark:text-white mb-2">

   542→                  <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">

   578→                              </div>
   579→                            </CardContent>

   596→                    {language === "ar" ? <ChevronRight className="w-4 h-4" /> : <ChevronLeft className="w-4 h-4" />}

   603→                    </h2>
   604→                  </div>
   605→
   606→                  {/* الوحدات */}
   607→                  {selectedSubject.units?.map((unit, unitIndex) => (

   620→                <motion.div
   621→                  initial={{ opacity: 0, y: 20 }}
   622→                  animate={{ opacity: 1, y: 0 }}
   623→                  transition={{ delay: 0.5 }}
   624→                  className="mt-8 grid grid-cols-2 md:grid-cols-4 gap-4"
   625→                >
   626→                  <Card className="text-center p-4">
   627→                    <Database className="w-8 h-8 mx-auto mb-2 text-emerald-500" />
   628→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">3</div>
   629→                    <div className="text-sm text-slate-500">{language === "ar" ? "سنوات دراسية" : "Years"}</div>
   630→                  </Card>
   631→                  <Card className="text-center p-4">
   632→                    <BookOpen className="w-8 h-8 mx-auto mb-2 text-blue-500" />

   640→                  </Card>
   641→                  <Card className="text-center p-4">
   642→                    <Brain className="w-8 h-8 mx-auto mb-2 text-amber-500" />
   643→                    <div className="text-2xl font-bold text-slate-800 dark:text-white">{points}</div>
   644→                    <div className="text-sm text-slate-500">{language === "ar" ? "نقطة" : "Points"}</div>
   645→                  </Card>
   646→                </motion.div>
   647→              )}
   648→            </motion.div>
   649→          )}
   650→        </AnimatePresence>
   651→      </main>
   652→
   653→      {/* الفوتر */}
   654→      <footer className="mt-auto border-t border-slate-200 dark:border-slate-800 bg-white/80 dark:bg-slate-900/80 backdrop-blur-sm">
   655→        <div className="container mx-auto px-4 py-4">
   656→          <div className="flex flex-col md:flex-row items-center justify-between gap-4">
   657→            <p className="text-sm text-slate-500 dark:text-slate-400 text-center md:text-start">
   658→              {language === "ar" 
   659→                ? "© 2024 ثانوية تفاعلية - المنهج المصري الكامل" 
   660→                : "© 2024 Interactive HS - Complete Egyptian Curriculum"}
   661→            </p>
   662→            <div className="flex items-center gap-4">
   663→              <Badge variant="secondary" className="gap-1">
   664→                <Database className="w-3 h-3" />
   665→                API Ready
   666→              </Badge>
   667→              <Badge variant="secondary" className="gap-1">
   668→                <Sparkles className="w-3 h-3" />
   669→                v3.0
   670→              </Badge>
   671→            </div>
   672→          </div>
   673→        </div>
   674→      </footer>
   675→
   676→      {/* نافذة المكافآت */}
   677→      <RewardsSystem
   678→        isOpen={showRewards}
   679→        onClose={() => setShowRewards(false)}
   680→        userProgress={userProgress}
