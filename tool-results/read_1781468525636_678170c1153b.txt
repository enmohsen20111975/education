   354→   354→function CircleDiagram({
   355→   355→  data,
   356→   356→  language,
   357→   357→}: {
   358→   358→  data: CircleData;
   359→   359→  language: "ar" | "en";
   360→   360→}) {
   361→   361→  const dir = language === "ar" ? "rtl" : "ltr";
   362→   362→  const isRTL = language === "ar";
   363→   363→
   364→   364→  // حساب الزوايا والأقواس
   365→   365→  const total = data.segments.reduce((sum, seg) => sum + seg.value, 0);
   366→   366→  
   367→   367→  const segmentsWithAngles = useMemo(() => {
   368→   368→    // استخدام reduce لبناء المصفوفة مع تتبع الزاوية الحالية
   369→   369→    const result = data.segments.reduce<{
   370→   370→      segments: Array<{
   371→   371→        label: string;
   372→   372→        labelAr: string;
   373→   373→        value: number;
   374→   374→        color: string;
   375→   375→        description?: string;
   376→   376→        descriptionAr?: string;
   377→   377→        percentage: number;
   378→   378→        startAngle: number;
   379→   379→        endAngle: number;
   380→   380→      }>;
   381→   381→      currentAngle: number;
   382→   382→    }>((acc, seg, index) => {
   383→   383→      const percentage = (seg.value / total) * 100;
   384→   384→      const angle = (seg.value / total) * 360;
   385→   385→      const startAngle = acc.currentAngle;
   386→   386→      const endAngle = acc.currentAngle + angle;
   387→   387→
   388→   388→      acc.segments.push({
   389→   389→        ...seg,
   390→   390→        percentage,
   391→   391→        startAngle,
   392→   392→        endAngle,
   393→   393→        color: seg.color || SEGMENT_COLORS[index % SEGMENT_COLORS.length],
   394→   394→      });
   395→   395→
   396→   396→      acc.currentAngle = endAngle;
   397→   397→      return acc;
   398→   398→    }, { segments: [], currentAngle: -90 });
   399→   399→
   400→   400→    return result.segments;
   401→   401→  }, [data.segments, total]);
   402→   402→
   403→   403→  // إنشاء مسار القوس SVG
   404→   404→  const createArcPath = (startAngle: number, endAngle: number, radius: number) => {
   405→   405→    const startRad = (startAngle * Math.PI) / 180;
   406→   406→    const endRad = (endAngle * Math.PI) / 180;
   407→   407→    
   408→   408→    const startX = 100 + radius * Math.cos(startRad);
   409→   409→    const startY = 100 + radius * Math.sin(startRad);
   410→   410→    const endX = 100 + radius * Math.cos(endRad);
   411→   411→    const endY = 100 + radius * Math.sin(endRad);
   412→   412→    
   413→   413→    const largeArc = endAngle - startAngle > 180 ? 1 : 0;
   414→   414→    
   415→   415→    return `M 100 100 L ${startX} ${startY} A ${radius} ${radius} 0 ${largeArc} 1 ${endX} ${endY} Z`;
   416→   416→  };
   417→   417→
   418→   418→  return (
   419→   419→    <motion.div
   420→   420→      className="space-y-6"
   421→   421→      variants={containerVariants}
   422→   422→      initial="hidden"
   423→   423→      animate="visible"
   424→   424→      dir={dir}
   425→   425→    >
   426→   426→      <div className="flex flex-col lg:flex-row items-center gap-8">
   427→   427→        {/* الدائرة */}
   428→   428→        <motion.div
   429→   429→          variants={scaleVariants}
   430→   430→          className="relative w-64 h-64 lg:w-80 lg:h-80"
   431→   431→        >
   432→   432→          <svg viewBox="0 0 200 200" className="w-full h-full transform -rotate-0">
   433→   433→            {/* خلفية الدائرة */}
   434→   434→            <circle
   435→   435→              cx="100"
   436→   436→              cy="100"
   437→   437→              r="80"
   438→   438→              fill="none"
   439→   439→              stroke="currentColor"
   440→   440→              strokeWidth="2"
   441→   441→              className="text-slate-200 dark:text-slate-700"
   442→   442→            />
   443→   443→            
   444→   444→            {/* الأقسام */}
   445→   445→            {segmentsWithAngles.map((seg, index) => (
   446→   446→              <motion.path
   447→   447→                key={index}
   448→   448→                d={createArcPath(seg.startAngle, seg.endAngle, 80)}
   449→   449→                fill={seg.color}
   450→   450→                initial={{ opacity: 0, scale: 0.8 }}
   451→   451→                animate={{ opacity: 1, scale: 1 }}
   452→   452→                transition={{ duration: 0.5, delay: index * 0.1 }}
   453→   453→                className="hover:opacity-80 transition-opacity cursor-pointer"
   454→   454→              />
   455→   455→            ))}
   456→   456→            
   457→   457→            {/* الدائرة المركزية */}
   458→   458→            <circle
   459→   459→              cx="100"
   460→   460→              cy="100"
   461→   461→              r="40"
   462→   462→              fill="white"
   463→   463→              className="dark:fill-slate-800"
   464→   464→            />
   465→   465→            
   466→   466→            {/* النص المركزي */}
   467→   467→            <text
   468→   468→              x="100"
   469→   469→              y="100"
   470→   470→              textAnchor="middle"
   471→   471→              dominantBaseline="middle"
   472→   472→              className="text-sm font-bold fill-slate-700 dark:fill-slate-200"
   473→   473→            >
   474→   474→              {isRTL ? data.centerLabelAr : data.centerLabel || (data.showPercentage !== false ? `${total}%` : total)}
   475→   475→            </text>
   476→   476→          </svg>
   477→   477→        </motion.div>
   478→   478→
   479→   479→        {/* التسميات */}
   480→   480→        <div className="flex-1 space-y-3">
   481→   481→          {segmentsWithAngles.map((seg, index) => (
   482→   482→            <motion.div
   483→   483→              key={index}
   484→   484→              variants={itemVariants}
   485→   485→              className="flex items-center gap-3 p-3 bg-slate-50 dark:bg-slate-800 rounded-lg hover:shadow-md transition-shadow"
   486→   486→            >
   487→   487→              <div
   488→   488→                className="w-4 h-4 rounded-full shrink-0"
   489→   489→                style={{ backgroundColor: seg.color }}
   490→   490→              />
   491→   491→              <div className="flex-1">
   492→   492→                <div className="flex items-center justify-between">
   493→   493→                  <span className="font-medium">{isRTL ? seg.labelAr : seg.label}</span>
   494→   494→                  <Badge variant="outline" className="ml-2">
   495→   495→                    {data.showPercentage !== false ? `${seg.percentage.toFixed(1)}%` : seg.value}
