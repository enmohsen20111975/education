   134→   134→
   135→   135→function VideoFactoryContent() {
   136→   136→  const { language, toggleLanguage, t } = useLanguage();
   137→   137→  const { theme, setTheme } = useTheme();
   138→   138→  const isDark = theme === "dark";
   139→   139→
   140→   140→  // ─── حالة التطبيق / App State ───
   141→   141→  const [lessons, setLessons] = useState<LessonItem[]>([]);
   142→   142→  const [selectedLesson, setSelectedLesson] = useState<string>("");
   143→   143→  const [selectedLanguage, setSelectedLanguage] = useState<string>("ar");
   144→   144→  const [selectedStyle, setSelectedStyle] = useState<string>("explainer");
   145→   145→  const [selectedVoice, setSelectedVoice] = useState<string>("female-ar");
   146→   146→  const [jobs, setJobs] = useState<VideoJob[]>([]);
   147→   147→  const [activeJob, setActiveJob] = useState<VideoJob | null>(null);
   148→   148→  const [isProducing, setIsProducing] = useState(false);
   149→   149→  const [isExporting, setIsExporting] = useState(false);
   150→   150→  const [isLoadingLessons, setIsLoadingLessons] = useState(false);
   151→   151→  const [toast, setToast] = useState<{ message: string; type: "success" | "error" } | null>(null);
   152→   152→  const pollingRef = useRef<NodeJS.Timeout | null>(null);
   153→   153→
   154→   154→  // ─── تحميل الدروس / Load lessons ───
   155→   155→  const loadLessons = useCallback(async () => {
   156→   156→    setIsLoadingLessons(true);
   157→   157→    try {
   158→   158→      const res = await fetch("/api/lessons");
   159→   159→      if (res.ok) {
   160→   160→        const data = await res.json();
   161→   161→        const lessonItems: LessonItem[] = (data.data?.lessons ?? data.lessons ?? []).map(
   162→   162→          (l: { id: string; titleAr: string; titleEn: string; Unit?: { nameAr: string } }) => ({
   163→   163→            id: l.id,
   164→   164→            titleAr: l.titleAr,
   165→   165→            titleEn: l.titleEn,
   166→   166→            unitName: l.Unit?.nameAr,
   167→   167→          })
   168→   168→        );
   169→   169→        setLessons(lessonItems);
   170→   170→        if (lessonItems.length > 0 && !selectedLesson) {
   171→   171→          setSelectedLesson(lessonItems[0].id);
   172→   172→        }
   173→   173→      }
   174→   174→    } catch {
   175→   175→      console.error("Failed to load lessons");
   176→   176→    } finally {
   177→   177→      setIsLoadingLessons(false);
   178→   178→    }
   179→   179→  }, [selectedLesson]);
   180→   180→
   181→   181→  // ─── تحميل المهام / Load jobs ───
   182→   182→  const loadJobs = useCallback(async () => {
   183→   183→    try {
   184→   184→      const res = await fetch("/api/video/jobs");
   185→   185→      if (res.ok) {
   186→   186→        const data = await res.json();
   187→   187→        setJobs(data.data?.jobs ?? []);
   188→   188→        // تحديث المهمة النشطة
   189→   189→        const activeJobs = (data.data?.jobs ?? []).filter(
   190→   190→          (j: VideoJob) => !["done", "error", "cancelled"].includes(j.status)
   191→   191→        );
   192→   192→        if (activeJobs.length > 0) {
   193→   193→          setActiveJob(activeJobs[0]);
   194→   194→          setIsProducing(true);
   195→   195→        } else {
   196→   196→          if (activeJob && (activeJob.status === "done" || activeJob.status === "error")) {
   197→   197→            // Keep showing completed job briefly
   198→   198→          } else {
   199→   199→            setActiveJob(null);
   200→   200→            setIsProducing(false);
   201→   201→          }
   202→   202→        }
   203→   203→      }
   204→   204→    } catch {
   205→   205→      console.error("Failed to load jobs");
   206→   206→    }
   207→   207→  }, [activeJob]);
   208→   208→
   209→   209→  // ─── جلب البيانات الأولية / Fetch initial data ───
   210→   210→  useEffect(() => {
   211→   211→    loadLessons();
   212→   212→    loadJobs();
   213→   213→  }, [loadLessons, loadJobs]);
   214→   214→
   215→   215→  // ─── استقصاء المهام النشطة / Poll active jobs ───
   216→   216→  useEffect(() => {
   217→   217→    if (isProducing) {
   218→   218→      pollingRef.current = setInterval(loadJobs, 2000);
   219→   219→    } else if (pollingRef.current) {
   220→   220→      clearInterval(pollingRef.current);
   221→   221→      pollingRef.current = null;
   222→   222→    }
   223→   223→    return () => {
   224→   224→      if (pollingRef.current) clearInterval(pollingRef.current);
   225→   225→    };
   226→   226→  }, [isProducing, loadJobs]);
   227→   227→
   228→   228→  // ─── إظهار إشعار / Show toast ───
   229→   229→  useEffect(() => {
   230→   230→    if (toast) {
   231→   231→      const timer = setTimeout(() => setToast(null), 4000);
   232→   232→      return () => clearTimeout(timer);
   233→   233→    }
   234→   234→  }, [toast]);
   235→   235→
   236→   236→  // ─── بدء الإنتاج / Start production ───
   237→   237→  const handleProduce = async () => {
   238→   238→    if (!selectedLesson) return;
   239→   239→
   240→   240→    try {
   241→   241→      setIsProducing(true);
   242→   242→      const res = await fetch("/api/video/produce", {
   243→   243→        method: "POST",
   244→   244→        headers: { "Content-Type": "application/json" },
   245→   245→        body: JSON.stringify({
   246→   246→          lessonId: selectedLesson,
   247→   247→          language: selectedLanguage,
   248→   248→          style: selectedStyle,
   249→   249→          voice: selectedVoice,
   250→   250→        }),
   251→   251→      });
   252→   252→
   253→   253→      const data = await res.json();
   254→   254→      if (data.success) {
   255→   255→        setToast({ message: data.data?.message || "تم إنشاء المهمة بنجاح", type: "success" });
   256→   256→        loadJobs();
   257→   257→      } else {
   258→   258→        setToast({ message: data.error || "حدث خطأ", type: "error" });
   259→   259→        setIsProducing(false);
   260→   260→      }
   261→   261→    } catch {
   262→   262→      setToast({ message: "فشل الاتصال بالخادم", type: "error" });
   263→   263→      setIsProducing(false);
   264→   264→    }
   265→   265→  };
   266→   266→
   267→   267→  // ─── إلغاء المهمة / Cancel job ───
   268→   268→  const handleCancel = async (jobId: string) => {
   269→   269→    try {
   270→   270→      const res = await fetch("/api/video/cancel", {
   271→   271→        method: "POST",
   272→   272→        headers: { "Content-Type": "application/json" },
   273→   273→        body: JSON.stringify({ jobId }),
   274→   274→      });
   275→   275→
   276→   276→      const data = await res.json();
   277→   277→      if (data.success) {
   278→   278→        setToast({ message: "تم إلغاء المهمة", type: "success" });
   279→   279→        loadJobs();
   280→   280→      }
   281→   281→    } catch {
   282→   282→      setToast({ message: "فشل إلغاء المهمة", type: "error" });
   283→   283→    }
   284→   284→  };
   285→   285→
   286→   286→  // ─── تصدير البيانات / Export data ───
   287→   287→  const handleExport = async (lessonId: string) => {
   288→   288→    setIsExporting(true);

   290→   290→      const res = await fetch(`/api/video/export/${lessonId}?language=${selectedLanguage}`);
   291→   291→      if (res.ok) {
   292→   292→        const data = await res.json();
   293→   293→        const blob = new Blob([JSON.stringify(data.data, null, 2)], { type: "application/json" });
   294→   294→        const url = URL.createObjectURL(blob);
   295→   295→        const a = document.createElement("a");
   296→   296→        a.href = url;
   297→   297→        a.download = `remotion-export-${lessonId}.json`;
   298→   298→        a.click();
   299→   299→        URL.revokeObjectURL(url);
   300→   300→        setToast({ message: "تم تصدير البيانات بنجاح", type: "success" });
   301→   301→      } else {
   302→   302→        const data = await res.json();
   303→   303→        setToast({ message: data.error || "فشل التصدير", type: "error" });
   304→   304→      }
   305→   305→    } catch {
   306→   306→      setToast({ message: "فشل التصدير", type: "error" });
   307→   307→    } finally {
   308→   308→      setIsExporting(false);
   309→   309→    }
   310→   310→  };
   311→   311→
   312→   312→  const selectedLessonData = lessons.find((l) => l.id === selectedLesson);
   313→   313→
   314→   314→  return (
   315→   315→    <div
   316→   316→      className={`min-h-screen flex flex-col bg-background overflow-x-hidden ${language === "ar" ? "rtl" : "ltr"}`}
   317→   317→      dir={language === "ar" ? "rtl" : "ltr"}
   318→   318→    >
   319→   319→      {/* ==================== NAVBAR ==================== */}
   320→   320→      <motion.nav
   321→   321→        initial={{ y: -80 }}
   322→   322→        animate={{ y: 0 }}
   323→   323→        transition={{ duration: 0.4 }}
   324→   324→        className="sticky top-0 z-50 border-b border-border/40 backdrop-blur-xl bg-background/80"
   325→   325→      >
   326→   326→        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
   327→   327→          <div className="flex items-center gap-3">
   328→   328→            <motion.div
   329→   329→              className="w-10 h-10 rounded-xl bg-gradient-to-br from-purple-600 to-pink-600 flex items-center justify-center"
   330→   330→              whileHover={{ scale: 1.05 }}
   331→   331→            >
   332→   332→              <Clapperboard className="w-5 h-5 text-white" />
   333→   333→            </motion.div>
   334→   334→            <div>
   335→   335→              <span className="font-bold text-lg bg-gradient-to-r from-purple-600 to-pink-600 bg-clip-text text-transparent">
   336→   336→                {t("مصنع الفيديو", "Video Factory")}
   337→   337→              </span>
   338→   338→              <p className="text-xs text-muted-foreground">
   339→   339→                {t("Phase 5 — خط إنتاج الفيديو التعليمي", "Phase 5 — Educational Video Pipeline")}
   340→   340→              </p>
   341→   341→            </div>
   342→   342→          </div>
   343→   343→
   344→   344→          <div className="flex items-center gap-2">
   345→   345→            <Button variant="ghost" size="sm" onClick={toggleLanguage} className="rounded-full">
   346→   346→              <Globe className="w-4 h-4 mr-1" />
   347→   347→              {language === "ar" ? "EN" : "عربي"}
   348→   348→            </Button>
   349→   349→            <Button variant="ghost" size="icon" onClick={() => setTheme(isDark ? "light" : "dark")} className="rounded-full">
   350→   350→              {isDark ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
   351→   351→            </Button>
   352→   352→          </div>
   353→   353→        </div>
   354→   354→      </motion.nav>
   355→   355→
   356→   356→      {/* ==================== TOAST ==================== */}
   357→   357→      <AnimatePresence>
   358→   358→        {toast && (
   359→   359→          <motion.div
   360→   360→            initial={{ opacity: 0, y: -20, x: "50%" }}
   361→   361→            animate={{ opacity: 1, y: 0, x: "50%" }}
   362→   362→            exit={{ opacity: 0, y: -20 }}
   363→   363→            className={`fixed top-20 left-1/2 -translate-x-1/2 z-[100] px-4 py-2.5 rounded-xl border shadow-lg text-sm font-medium backdrop-blur-xl ${
   364→   364→              toast.type === "success"
   365→   365→                ? "bg-emerald-500/15 border-emerald-500/30 text-emerald-600"
   366→   366→                : "bg-red-500/15 border-red-500/30 text-red-600"
   367→   367→            }`}
   368→   368→          >
   369→   369→            {toast.message}
   370→   370→          </motion.div>
   371→   371→        )}
   372→   372→      </AnimatePresence>
   373→   373→
   374→   374→      {/* ==================== MAIN CONTENT ==================== */}
   375→   375→      <main className="flex-1 max-w-7xl mx-auto px-4 py-6 w-full">
   376→   376→        <motion.div initial="hidden" animate="visible" variants={staggerContainer} className="space-y-6">
   377→   377→          {/* ─── Hero / Stats ─── */}
   378→   378→          <motion.div variants={fadeInUp} className="grid grid-cols-2 md:grid-cols-4 gap-4">
   379→   379→            {[
   380→   380→              {
   381→   381→                icon: Film,
   382→   382→                value: jobs.filter((j) => j.status === "done").length.toString(),
   383→   383→                label: { ar: "فيديو مكتمل", en: "Videos Done" },
   384→   384→                color: "from-emerald-500 to-teal-500",
   385→   385→              },
   386→   386→              {
   387→   387→                icon: Loader2,
   388→   388→                value: jobs.filter((j) => ["queued", "scripting", "tts", "composing", "rendering"].includes(j.status)).length.toString(),
   389→   389→                label: { ar: "قيد الإنتاج", en: "In Progress" },
   390→   390→                color: "from-purple-500 to-pink-500",

   421→   421→          </motion.div>
