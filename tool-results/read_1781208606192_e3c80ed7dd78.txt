   271→function AnimatedCounter({ 
   272→  end, 
   273→  duration = 2, 
   274→  suffix = "" 
   275→}: { 
   276→  end: number; 
   277→  duration?: number; 
   278→  suffix?: string;
   279→}) {
   280→  const [count, setCount] = useState(0);
   281→  const [isVisible, setIsVisible] = useState(false);
   282→  const ref = useRef<HTMLSpanElement>(null);
   283→
   284→  useEffect(() => {
   285→    const observer = new IntersectionObserver(
   286→      ([entry]) => {
   287→        if (entry.isIntersecting) {
   288→          setIsVisible(true);
   289→        }
   290→      },
   291→      { threshold: 0.1 }
   292→    );
   293→
   294→    if (ref.current) {
   295→      observer.observe(ref.current);
   296→    }
   297→
   298→    return () => observer.disconnect();
   299→  }, []);
   300→
   301→  useEffect(() => {
   302→    if (!isVisible) return;
   303→
   304→    let startTime: number;
   305→    const animate = (currentTime: number) => {
   306→      if (!startTime) startTime = currentTime;
   307→      const progress = Math.min((currentTime - startTime) / (duration * 1000), 1);
   308→      
   309→      setCount(Math.floor(progress * end));
   310→
   311→      if (progress < 1) {
   312→        requestAnimationFrame(animate);
   313→      }
   314→    };
   315→
   316→    requestAnimationFrame(animate);
   317→  }, [isVisible, end, duration]);
   318→
   319→  return (
   320→    <span ref={ref} className="tabular-nums">
   321→      {count}{suffix}
   322→    </span>
   323→  );
   324→}
