   290→// ===== مكون العداد المتحرك =====
   291→function AnimatedCounter({ 
   292→  end, 
   293→  duration = 2, 
   294→  suffix = "" 
   295→}: { 
   296→  end: number; 
   297→  duration?: number; 
   298→  suffix?: string;
   299→}) {
   300→  const [count, setCount] = useState(0);
   301→  const [isVisible, setIsVisible] = useState(false);
   302→  const ref = useRef<HTMLSpanElement>(null);
   303→
   304→  useEffect(() => {
   305→    const observer = new IntersectionObserver(
   306→      ([entry]) => {
   307→        if (entry.isIntersecting) {
   308→          setIsVisible(true);
   309→        }
   310→      },
   311→      { threshold: 0.1 }
   312→    );
   313→
   314→    if (ref.current) {
   315→      observer.observe(ref.current);
   316→    }
   317→
   318→    return () => observer.disconnect();
   319→  }, []);
   320→
   321→  useEffect(() => {
   322→    if (!isVisible) return;
   323→
   324→    let startTime: number;
   325→    const animate = (currentTime: number) => {
   326→      if (!startTime) startTime = currentTime;
   327→      const progress = Math.min((currentTime - startTime) / (duration * 1000), 1);
   328→      
   329→      setCount(Math.floor(progress * end));
   330→
   331→      if (progress < 1) {
   332→        requestAnimationFrame(animate);
   333→      }
   334→    };
   335→
   336→    requestAnimationFrame(animate);
   337→  }, [isVisible, end, duration]);
   338→
   339→  return (
   340→    <span ref={ref} className="tabular-nums">
   341→      {count}{suffix}
   342→    </span>
   343→  );
   344→}
