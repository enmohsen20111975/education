   134→
   135→function VideoFactoryContent() {
   136→  const { language, toggleLanguage, t } = useLanguage();
   137→  const { theme, setTheme } = useTheme();
   138→  const isDark = theme === "dark";
   139→
   140→  // ─── حالة التطبيق / App State ───
   141→  const [lessons, setLessons] = useState<LessonItem[]>([]);
   142→  const [selectedLesson, setSelectedLesson] = useState<string>("");
   143→  const [selectedLanguage, setSelectedLanguage] = useState<string>("ar");
   144→  const [selectedStyle, setSelectedStyle] = useState<string>("explainer");
   145→  const [selectedVoice, setSelectedVoice] = useState<string>("female-ar");
   146→  const [jobs, setJobs] = useState<VideoJob[]>([]);
   147→  const [activeJob, setActiveJob] = useState<VideoJob | null>(null);
   148→  const [isProducing, setIsProducing] = useState(false);
   149→  const [isExporting, setIsExporting] = useState(false);
   150→  const [isLoadingLessons, setIsLoadingLessons] = useState(false);
   151→  const [toast, setToast] = useState<{ message: string; type: "success" | "error" } | null>(null);
   152→  const pollingRef = useRef<NodeJS.Timeout | null>(null);
   153→
   154→  // ─── تحميل الدروس / Load lessons ───
   155→  const loadLessons = useCallback(async () => {
   156→    setIsLoadingLessons(true);
   157→    try {
   158→      const res = await fetch("/api/lessons");
   159→      if (res.ok) {
   160→        const data = await res.json();
   161→        const lessonItems: LessonItem[] = (data.data?.lessons ?? data.lessons ?? []).map(
   162→          (l: { id: string; titleAr: string; titleEn: string; Unit?: { nameAr: string } }) => ({
   163→            id: l.id,
   164→            titleAr: l.titleAr,
   165→            titleEn: l.titleEn,
   166→            unitName: l.Unit?.nameAr,
   167→          })
   168→        );
   169→        setLessons(lessonItems);
   170→        if (lessonItems.length > 0 && !selectedLesson) {
   171→          setSelectedLesson(lessonItems[0].id);
   172→        }
   173→      }
   174→    } catch {
   175→      console.error("Failed to load lessons");
   176→    } finally {
   177→      setIsLoadingLessons(false);
   178→    }
   179→  }, [selectedLesson]);
   180→
   181→  // ─── تحميل المهام / Load jobs ───
   182→  const loadJobs = useCallback(async () => {
   183→    try {
   184→      const res = await fetch("/api/video/jobs");
   185→      if (res.ok) {
   186→        const data = await res.json();
   187→        setJobs(data.data?.jobs ?? []);
   188→        // تحديث المهمة النشطة
   189→        const activeJobs = (data.data?.jobs ?? []).filter(
   190→          (j: VideoJob) => !["done", "error", "cancelled"].includes(j.status)
   191→        );
   192→        if (activeJobs.length > 0) {
   193→          setActiveJob(activeJobs[0]);
   194→          setIsProducing(true);
   195→        } else {
   196→          if (activeJob && (activeJob.status === "done" || activeJob.status === "error")) {
   197→            // Keep showing completed job briefly
   198→          } else {
   199→            setActiveJob(null);
   200→            setIsProducing(false);
   201→          }
   202→        }
   203→      }
   204→    } catch {
   205→      console.error("Failed to load jobs");
   206→    }
   207→  }, [activeJob]);
   208→
   209→  // ─── جلب البيانات الأولية / Fetch initial data ───
   210→  useEffect(() => {
   211→    loadLessons();
   212→    loadJobs();
   213→  }, [loadLessons, loadJobs]);
   214→
   215→  // ─── استقصاء المهام النشطة / Poll active jobs ───
   216→  useEffect(() => {
   217→    if (isProducing) {
   218→      pollingRef.current = setInterval(loadJobs, 2000);
   219→    } else if (pollingRef.current) {
   220→      clearInterval(pollingRef.current);
   221→      pollingRef.current = null;
   222→    }
   223→    return () => {
   224→      if (pollingRef.current) clearInterval(pollingRef.current);
   225→    };
   226→  }, [isProducing, loadJobs]);
   227→
   228→  // ─── إظهار إشعار / Show toast ───
   229→  useEffect(() => {
   230→    if (toast) {
   231→      const timer = setTimeout(() => setToast(null), 4000);
   232→      return () => clearTimeout(timer);
   233→    }
   234→  }, [toast]);
   235→
   236→  // ─── بدء الإنتاج / Start production ───
   237→  const handleProduce = async () => {
   238→    if (!selectedLesson) return;
   239→
   240→    try {
   241→      setIsProducing(true);
   242→      const res = await fetch("/api/video/produce", {
   243→        method: "POST",
   244→        headers: { "Content-Type": "application/json" },
   245→        body: JSON.stringify({
   246→          lessonId: selectedLesson,
   247→          language: selectedLanguage,
   248→          style: selectedStyle,
   249→          voice: selectedVoice,
   250→        }),
   251→      });
   252→
   253→      const data = await res.json();
   254→      if (data.success) {
   255→        setToast({ message: data.data?.message || "تم إنشاء المهمة بنجاح", type: "success" });
   256→        loadJobs();
   257→      } else {
   258→        setToast({ message: data.error || "حدث خطأ", type: "error" });
   259→        setIsProducing(false);
   260→      }
   261→    } catch {
   262→      setToast({ message: "فشل الاتصال بالخادم", type: "error" });
   263→      setIsProducing(false);
   264→    }
   265→  };
   266→
   267→  // ─── إلغاء المهمة / Cancel job ───
   268→  const handleCancel = async (jobId: string) => {
   269→    try {
   270→      const res = await fetch("/api/video/cancel", {
   271→        method: "POST",
   272→        headers: { "Content-Type": "application/json" },
   273→        body: JSON.stringify({ jobId }),
   274→      });
   275→
   276→      const data = await res.json();
   277→      if (data.success) {
   278→        setToast({ message: "تم إلغاء المهمة", type: "success" });
   279→        loadJobs();
   280→      }
   281→    } catch {
   282→      setToast({ message: "فشل إلغاء المهمة", type: "error" });
   283→    }
   284→  };
   285→
   286→  // ─── تصدير البيانات / Export data ───
   287→  const handleExport = async (lessonId: string) => {
   288→    setIsExporting(true);
   289→    try {
   290→      const res = await fetch(`/api/video/export/${lessonId}?language=${selectedLanguage}`);
   291→      if (res.ok) {
   292→        const data = await res.json();
   293→        const blob = new Blob([JSON.stringify(data.data, null, 2)], { type: "application/json" });
   294→        const url = URL.createObjectURL(blob);
   295→        const a = document.createElement("a");
   296→        a.href = url;
   297→        a.download = `remotion-export-${lessonId}.json`;
   298→        a.click();
   299→        URL.revokeObjectURL(url);
   300→        setToast({ message: "تم تصدير البيانات بنجاح", type: "success" });
   301→      } else {
   302→        const data = await res.json();
   303→        setToast({ message: data.error || "فشل التصدير", type: "error" });
   304→      }
   305→    } catch {
   306→      setToast({ message: "فشل التصدير", type: "error" });
   307→    } finally {
   308→      setIsExporting(false);
   309→    }
   310→  };
   311→
   312→  const selectedLessonData = lessons.find((l) => l.id === selectedLesson);
   313→
   314→  return (
   315→    <div
   316→      className={`min-h-screen flex flex-col bg-background overflow-x-hidden ${language === "ar" ? "rtl" : "ltr"}`}
   317→      dir={language === "ar" ? "rtl" : "ltr"}
   318→    >
   319→      {/* ==================== NAVBAR ==================== */}
   320→      <motion.nav
   321→        initial={{ y: -80 }}
   322→        animate={{ y: 0 }}
   323→        transition={{ duration: 0.4 }}
   324→        className="sticky top-0 z-50 border-b border-border/40 backdrop-blur-xl bg-background/80"
   325→      >
   326→        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
   327→          <div className="flex items-center gap-3">
   328→            <motion.div
   329→              className="w-10 h-10 rounded-xl bg-gradient-to-br from-purple-600 to-pink-600 flex items-center justify-center"
   330→              whileHover={{ scale: 1.05 }}
   331→            >
   332→              <Clapperboard className="w-5 h-5 text-white" />
   333→            </motion.div>
   334→            <div>
   335→              <span className="font-bold text-lg bg-gradient-to-r from-purple-600 to-pink-600 bg-clip-text text-transparent">
   336→                {t("مصنع الفيديو", "Video Factory")}
   337→              </span>
   338→              <p className="text-xs text-muted-foreground">
   339→                {t("Phase 5 — خط إنتاج الفيديو التعليمي", "Phase 5 — Educational Video Pipeline")}
   340→              </p>
   341→            </div>
   342→          </div>
   343→
   344→          <div className="flex items-center gap-2">
   345→            <Button variant="ghost" size="sm" onClick={toggleLanguage} className="rounded-full">
   346→              <Globe className="w-4 h-4 mr-1" />
   347→              {language === "ar" ? "EN" : "عربي"}
   348→            </Button>
   349→            <Button variant="ghost" size="icon" onClick={() => setTheme(isDark ? "light" : "dark")} className="rounded-full">
   350→              {isDark ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
   351→            </Button>
   352→          </div>
   353→        </div>
   354→      </motion.nav>
   355→
   356→      {/* ==================== TOAST ==================== */}
   357→      <AnimatePresence>
   358→        {toast && (
   359→          <motion.div
   360→            initial={{ opacity: 0, y: -20, x: "50%" }}
   361→            animate={{ opacity: 1, y: 0, x: "50%" }}
   362→            exit={{ opacity: 0, y: -20 }}
   363→            className={`fixed top-20 left-1/2 -translate-x-1/2 z-[100] px-4 py-2.5 rounded-xl border shadow-lg text-sm font-medium backdrop-blur-xl ${
   364→              toast.type === "success"
   365→                ? "bg-emerald-500/15 border-emerald-500/30 text-emerald-600"
   366→                : "bg-red-500/15 border-red-500/30 text-red-600"
   367→            }`}
   368→          >
   369→            {toast.message}
   370→          </motion.div>
   371→        )}
   372→      </AnimatePresence>
   373→
   374→      {/* ==================== MAIN CONTENT ==================== */}
   375→      <main className="flex-1 max-w-7xl mx-auto px-4 py-6 w-full">
   376→        <motion.div initial="hidden" animate="visible" variants={staggerContainer} className="space-y-6">
   377→          {/* ─── Hero / Stats ─── */}
   378→          <motion.div variants={fadeInUp} className="grid grid-cols-2 md:grid-cols-4 gap-4">
   379→            {[
   380→              {
   381→                icon: Film,
   382→                value: jobs.filter((j) => j.status === "done").length.toString(),
   383→                label: { ar: "فيديو مكتمل", en: "Videos Done" },
   384→                color: "from-emerald-500 to-teal-500",
   385→              },
   386→              {
   387→                icon: Loader2,
   388→                value: jobs.filter((j) => ["queued", "scripting", "tts", "composing", "rendering"].includes(j.status)).length.toString(),
   389→                label: { ar: "قيد الإنتاج", en: "In Progress" },
   390→                color: "from-purple-500 to-pink-500",
   391→              },
   392→              {
   393→                icon: FileText,
   394→                value: jobs.filter((j) => j.status === "scripting").length.toString(),
   395→                label: { ar: "نصوص جاري توليدها", en: "Scripts Running" },
   396→                color: "from-cyan-500 to-blue-500",
   397→              },
   398→              {
   399→                icon: Video,
   400→                value: jobs.length.toString(),
   401→                label: { ar: "إجمالي المهام", en: "Total Jobs" },
   402→                color: "from-orange-500 to-red-500",
   403→              },
   404→            ].map((stat, i) => (
   405→              <motion.div key={i} variants={scaleIn} whileHover={{ y: -3 }}>
   406→                <Card className="border-0 shadow-md overflow-hidden">
   407→                  <CardContent className="p-4">
   408→                    <div className="flex items-center gap-3">
   409→                      <div className={`w-10 h-10 rounded-xl bg-gradient-to-r ${stat.color} flex items-center justify-center shrink-0`}>
   410→                        <stat.icon className="w-5 h-5 text-white" />
   411→                      </div>
   412→                      <div>
   413→                        <div className="text-2xl font-black">{stat.value}</div>
   414→                        <div className="text-xs text-muted-foreground">{t(stat.label.ar, stat.label.en)}</div>
   415→                      </div>
   416→                    </div>
   417→                  </CardContent>
   418→                </Card>
   419→              </motion.div>
   420→            ))}
   421→          </motion.div>
   422→
   423→          <div className="grid lg:grid-cols-3 gap-6">
   424→            {/* ==================== LEFT PANEL: Controls ==================== */}
   425→            <div className="lg:col-span-1 space-y-4">
   426→              {/* ─── Lesson Selector ─── */}
   427→              <motion.div variants={fadeInUp}>
   428→                <Card className="border shadow-sm">
   429→                  <CardHeader className="pb-3">
   430→                    <CardTitle className="text-base flex items-center gap-2">
   431→                      <Film className="w-4 h-4 text-purple-500" />
   432→                      {t("اختر درساً", "Select a Lesson")}
   433→                    </CardTitle>
   434→                    <CardDescription className="text-xs">
   435→                      {t("اختر الدرس لإنتاج فيديو تعليمي له", "Choose a lesson to produce an educational video")}
   436→                    </CardDescription>
   437→                  </CardHeader>
   438→                  <CardContent className="space-y-3">
   439→                    {isLoadingLessons ? (
   440→                      <div className="space-y-2">
   441→                        <Skeleton className="h-10 w-full" />
   442→                        <Skeleton className="h-10 w-full" />
   443→                        <Skeleton className="h-10 w-full" />
   444→                      </div>
   445→                    ) : (
   446→                      <Select value={selectedLesson} onValueChange={setSelectedLesson}>
   447→                        <SelectTrigger className="w-full">
   448→                          <SelectValue placeholder={t("اختر درساً...", "Choose a lesson...")} />
   449→                        </SelectTrigger>
   450→                        <SelectContent>
   451→                          <ScrollArea className="max-h-64">
   452→                            {lessons.map((lesson) => (
   453→                              <SelectItem key={lesson.id} value={lesson.id} className="py-2">
   454→                                <span className="font-medium">{language === "ar" ? lesson.titleAr : lesson.titleEn}</span>
   455→                                {lesson.unitName && (
   456→                                  <span className="text-muted-foreground text-xs mr-2">— {lesson.unitName}</span>
   457→                                )}
   458→                              </SelectItem>
   459→                            ))}
   460→                          </ScrollArea>
   461→                        </SelectContent>
   462→                      </Select>
   463→                    )}
   464→
   465→                    {selectedLessonData && (
   466→                      <div className="p-3 rounded-lg bg-muted/50 border border-border/50">
   467→                        <p className="text-sm font-semibold">{language === "ar" ? selectedLessonData.titleAr : selectedLessonData.titleEn}</p>
   468→                        {selectedLessonData.unitName && (
   469→                          <p className="text-xs text-muted-foreground">{selectedLessonData.unitName}</p>
   470→                        )}
   471→                      </div>
   472→                    )}
   473→                  </CardContent>
   474→                </Card>
   475→              </motion.div>
   476→
   477→              {/* ─── Production Settings ─── */}
   478→              <motion.div variants={fadeInUp}>
   479→                <Card className="border shadow-sm">
   480→                  <CardHeader className="pb-3">
   481→                    <CardTitle className="text-base flex items-center gap-2">
   482→                      <Settings className="w-4 h-4 text-orange-500" />
   483→                      {t("إعدادات الإنتاج", "Production Settings")}
   484→                    </CardTitle>
   485→                  </CardHeader>
   486→                  <CardContent className="space-y-4">
   487→                    {/* Language */}
   488→                    <div className="space-y-1.5">
   489→                      <label className="text-xs font-medium text-muted-foreground">
   490→                        {t("اللغة", "Language")}
   491→                      </label>

   493→                        <SelectTrigger className="w-full">
   494→                          <SelectValue />

   496→                        <SelectContent>
   497→                          <SelectItem value="ar">{t("العربية", "Arabic")}</SelectItem>

   499→                        </SelectContent>
   500→                      </Select>
   501→                    </div>
   502→
   503→                    {/* Style */}
   504→                    <div className="space-y-1.5">
