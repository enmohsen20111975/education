     1→/**

     4→ * Express + Socket.io REST API server.

     7→ *   - /api/books/...           Books CRUD + extraction control
     8→ *   - /api/books/:bookId/lessons/...  Lessons CRUD + images + review
     9→ *   - /api/videos/...          Video generation queue + streaming
    10→ *   - /api/config              Pipeline config
    11→ *   - /api/system/status       System resources (CPU/RAM/Disk/GPU)
    12→ *   - /data/*                  Static images, videos and PDFs
    13→ *
    14→ * Realtime events are emitted via socket.io:
    15→ *   - extraction-progress
    16→ *   - video-progress
    17→ *   - queue-update
    18→ *   - log

    92→const pdfStorage = multer.diskStorage({
    93→  destination: (req, _file, cb) => {
    94→    const bookId = req.params.bookId || req.body.bookId;
    95→    if (bookId) {
    96→      const dir = path.join(BOOKS_DIR, bookId);

    99→    } else {
   100→      // tmp dir for new uploads where bookId not known yet
   101→      const tmp = path.join(BOOKS_DIR, '_uploads');
   102→      fs.mkdirSync(tmp, { recursive: true });
   103→      cb(null, tmp);
   104→    }
   105→  },
   106→  filename: (_req, file, cb) => {
   107→    cb(null, 'source.pdf');
   108→  },
   109→});

   111→const imageStorage = multer.diskStorage({
   112→  destination: (req, _file, cb) => {

   114→    const dir = path.join(BOOKS_DIR, bookId, 'images', lessonId);
   115→    fs.mkdirSync(dir, { recursive: true });
   116→    cb(null, dir);
   117→  },

   119→    const id = crypto.randomBytes(4).toString('hex');

   121→    cb(null, `img-${id}${ext}`);

   132→      cb(new Error('Only PDF files are allowed'));

   149→// ------------------------------------------------------------------
   150→// Response helpers
   151→// ------------------------------------------------------------------
   152→function ok(res, data, code = 200) {
   153→  return res.status(code).json({ success: true, data });
   154→}
   155→function fail(res, message, code = 400) {
   156→  return res.status(code).json({ success: false, error: message });
   157→}
   158→function notFound(res, message = 'Not found') {
   159→  return res.status(404).json({ success: false, error: message });
   160→}

   162→// async handler wrapper to catch errors
   163→function wrap(fn) {
   164→  return (req, res, next) => Promise.resolve(fn(req, res, next)).catch((err) => {
   165→    console.error(`[ERROR] ${req.method} ${req.originalUrl}:`, err.message);
   166→    return res.status(500).json({ success: false, error: err.message || 'Internal server error' });
   167→  });
   168→}

   173→function getPdfPageCount(pdfPath) {
   174→  try {
   175→    if (pdfPath && fs.existsSync(pdfPath)) {
   176→      const out = execSync(`pdfinfo "${pdfPath}" 2>/dev/null`, { encoding: 'utf-8' });
   177→      const m = out.match(/Pages:\s+(\d+)/i);
   178→      if (m) return parseInt(m[1], 10);
   179→    }
   180→  } catch (_) { /* ignore */ }
   181→  return 0;
   182→}

   184→function getBookOr404(res, bookId) {
   185→  const master = books.getBook(bookId);
   186→  if (!master) {
   187→    notFound(res, `Book '${bookId}' not found`);
   188→    return null;
   189→  }
   190→  return master;
   191→}

   193→function getLessonOr404(res, bookId, lessonId) {

   196→    notFound(res, `Lesson '${lessonId}' not found in book '${bookId}'`);
   197→    return null;
   198→  }
   199→  return lesson;
   200→}

   212→// POST /api/books/upload - upload PDF, create book
   213→app.post('/api/books/upload', (req, res, next) => {
   214→  // Pre-parse: multer saves to BOOKS_DIR/_uploads/source.pdf since bookId
   215→  // is not known until we generate it from the title.
   216→  uploadPdf.single('pdf')(req, res, (err) => {
   217→    if (err) return fail(res, err.message || 'Upload failed', 400);
   218→    next();
   219→  });
   220→}, wrap(async (req, res) => {
   221→  if (!req.file) return fail(res, 'No PDF file uploaded (field name must be "pdf")', 400);
   222→
   223→  const title = req.body.title || path.basename(req.file.originalname, '.pdf');
   224→  const subject = req.body.subject || 'other';
   225→  const grade = req.body.grade || '3rd-secondary';
   226→  const publisher = req.body.publisher || '';
   227→
   228→  // Generate a unique book id (collision-safe)
   229→  let bookId = books.generateBookId(title);
   230→  while (books.bookExists(bookId)) {
   231→    bookId = books.generateBookId(title);
   232→  }
   233→
   234→  // Create master.json + book directory structure (lessons/, images/, videos/)
   235→  // createBook will mkdir the book directory itself.
   236→  const master = books.createBook({
   237→    id: bookId,
   238→    title,
   239→    subject,
   240→    grade,
   241→    publisher,
   242→    source_pdf: `books/${bookId}/source.pdf`,
   243→    total_pages: 0,
   244→  });
   245→
   246→  // Now move the uploaded PDF into the freshly-created book directory.
   247→  const finalPath = path.join(BOOKS_DIR, bookId, 'source.pdf');
   248→  if (req.file.path !== finalPath) {
   249→    fs.renameSync(req.file.path, finalPath);
   250→  }
   251→
   252→  // Inspect page count and update master
   253→  const totalPages = getPdfPageCount(finalPath);
   254→  master.book.total_pages = totalPages;
   255→  master.book.source_pdf = `books/${bookId}/source.pdf`;
   256→  books.saveBook(bookId, master);
   257→
   258→  io.emit('log', { level: 'info', msg: `Book '${title}' uploaded as ${bookId} (${totalPages} pages)` });
   259→  return ok(res, master, 201);
   260→}));

   277→// POST /api/books/:bookId/extract - start extraction
   278→app.post('/api/books/:bookId/extract', wrap((req, res) => {
   279→  const master = getBookOr404(res, req.params.bookId);
   280→  if (!master) return;
   281→
   282→  // Mark as extracting (actual python process is handled separately by
   283→  // content-extractor scripts; here we just flag the status so the UI
   284→  // can update. Real worker reads this status from master.json.)
   285→  books.updateExtractionStatus(req.params.bookId, 'extracting', 0);
   286→
   287→  // Attempt to spawn python extractor in background (best-effort).
   288→  try {
   289→    const extractorScript = path.join(ROOT_DIR, 'content-extractor', 'run-all.py');
   290→    if (fs.existsSync(extractorScript)) {
   291→      const py = spawn('python3', [extractorScript, '--book-id', req.params.bookId], {
   292→        cwd: ROOT_DIR,
   293→        detached: true,
   294→        stdio: 'ignore',
   295→      });
   296→      py.unref();
   297→      console.log(`[extract] spawned python pid=${py.pid} for ${req.params.bookId}`);
   298→    } else {
   299→      console.warn(`[extract] ${extractorScript} not found - skipping actual extraction`);
   300→    }
   301→  } catch (e) {
   302→    console.warn(`[extract] failed to spawn extractor: ${e.message}`);
   303→  }
   304→
   305→  io.emit('extraction-progress', { bookId: req.params.bookId, status: 'extracting', progress: 0 });
   306→  io.emit('log', { level: 'info', msg: `Extraction started for ${req.params.bookId}` });
   307→
   308→  return ok(res, { bookId: req.params.bookId, status: 'extracting', progress: 0 });
   309→}));

   336→// GET /api/books/:bookId/logs - read extraction.log
   337→app.get('/api/books/:bookId/logs', wrap((req, res) => {
   338→  const logPath = path.join(BOOKS_DIR, req.params.bookId, 'extraction.log');
   339→  if (!fs.existsSync(logPath)) {
   340→    return ok(res, { bookId: req.params.bookId, logs: '', exists: false });
   341→  }
   342→  // Limit output to last 1MB to avoid huge payloads
   343→  const stat = fs.statSync(logPath);
   344→  const maxBytes = 1024 * 1024;
   345→  let content;
   346→  if (stat.size > maxBytes) {
   347→    const fd = fs.openSync(logPath, 'r');
   348→    const buf = Buffer.alloc(maxBytes);
   349→    fs.readSync(fd, buf, 0, maxBytes, stat.size - maxBytes);
   350→    fs.closeSync(fd);
   351→    content = '... (truncated) ...\n' + buf.toString('utf-8');
   352→  } else {
   353→    content = fs.readFileSync(logPath, 'utf-8');
   354→  }
   355→  return ok(res, { bookId: req.params.bookId, logs: content, exists: true, size: stat.size });
   356→}));

   477→// POST /api/videos/generate-batch - body: { book_id, lesson_ids: [] }
   478→app.post('/api/videos/generate-batch', wrap((req, res) => {
   479→  const { book_id, lesson_ids } = req.body || {};
   480→  if (!book_id || !Array.isArray(lesson_ids)) {
   481→    return fail(res, 'Body must include {book_id, lesson_ids: []}', 400);
   482→  }
   483→  if (!books.bookExists(book_id)) return notFound(res, `Book '${book_id}' not found`);
   484→
   485→  const results = [];
   486→  for (const lessonId of lesson_ids) {
   487→    const lesson = lessons.getLesson(book_id, lessonId);
   488→    if (!lesson) {
   489→      results.push({ lessonId, ok: false, error: 'Lesson not found' });
   490→      continue;
   491→    }
   492→    const title = lesson.metadata?.title || lessonId;
   493→    const added = queue.addToQueue(book_id, lessonId, title);
   494→    if (added) {
   495→      lessons.updateVideoStatus(book_id, lessonId, { status: 'generating' });
   496→      results.push({ lessonId, ok: true, status: 'generating' });
   497→    } else {
   498→      results.push({ lessonId, ok: false, error: 'Already in queue' });
   499→    }
   500→  }
   501→  io.emit('queue-update', queue.getQueue());
   502→  io.emit('log', { level: 'info', msg: `Batch generate: ${results.filter(r => r.ok).length}/${lesson_ids.length} added` });
   503→  return ok(res, { book_id, results });
   504→}));

   509→  if (!lesson) return;

   651→function getGpuInfo() {
   652→  try {
   653→    const out = execSync('nvidia-smi --query-gpu=name,memory.total,memory.used,memory.free,utilization.gpu --format=csv,noheader,nounits', {
   654→      encoding: 'utf-8',
   655→      timeout: 3000,
   656→    }).trim();
   657→    const parts = out.split(',').map((s) => s.trim());
   658→    if (parts.length >= 5) {
   659→      return {
   660→        name: parts[0],
   661→        vram_total_mb: parseInt(parts[1], 10),
   662→        vram_used_mb: parseInt(parts[2], 10),
   663→        vram_free_mb: parseInt(parts[3], 10),
   664→        utilization_percent: parseInt(parts[4], 10),
   665→      };
   666→    }
   667→  } catch (_) { /* no GPU or nvidia-smi missing */ }
   668→  return null;
   669→}
   670→
   671→function getCpuPercent() {
   672→  // Average load over 1 minute, normalized by CPU count -> rough percent
   673→  const cpus = os.cpus().length || 1;
   674→  const load = os.loadavg()[0]; // 1-min avg
   675→  const percent = Math.min(100, Math.round((load / cpus) * 100));
   676→  return percent;
   677→}
   678→
   679→function getDiskUsage(dirPath) {
   680→  try {
   681→    const out = execSync(`df -k "${dirPath}" | tail -1`, { encoding: 'utf-8' });
   682→    const parts = out.trim().split(/\s+/);
   683→    if (parts.length >= 4) {
   684→      const total = parseInt(parts[1], 10) * 1024;
   685→      const used = parseInt(parts[2], 10) * 1024;
   686→      const free = parseInt(parts[3], 10) * 1024;
   687→      const percent = total > 0 ? Math.round((used / total) * 100) : 0;
   688→      return { total, used, free, percent };
   689→    }
   690→  } catch (_) { /* ignore */ }
   691→  return { total: 0, used: 0, free: 0, percent: 0 };
   692→}

   694→// GET /api/system/status
   695→app.get('/api/system/status', wrap((req, res) => {
   696→  const totalMem = os.totalmem();
   697→  const freeMem = os.freemem();
   698→  const usedMem = totalMem - freeMem;
   699→  const memPercent = totalMem > 0 ? Math.round((usedMem / totalMem) * 100) : 0;
   700→
   701→  const data = {
   702→    cpu_percent: getCpuPercent(),
   703→    cpu_cores: os.cpus().length,
   704→    cpu_model: os.cpus()[0]?.model || 'unknown',
   705→    memory: {
   706→      total: totalMem,
   707→      used: usedMem,
   708→      free: freeMem,
   709→      percent: memPercent,
   710→    },
   711→    disk: getDiskUsage(DATA_DIR),
   712→    uptime_sec: Math.round(os.uptime()),
   713→    platform: os.platform(),
   714→    hostname: os.hostname(),
   715→    node_version: process.version,
   716→    timestamp: new Date().toISOString(),
   717→  };
   718→
   719→  const gpu = getGpuInfo();
   720→  if (gpu) data.gpu = gpu;
   721→
   722→  return ok(res, data);
