    12→import {

   109→
   110→export default function LessonPageClient() {
   111→  const params = useParams();
   112→  const lessonId = params.id as string;
   113→  const { language, toggleLanguage, t } = useLanguage();
   114→  const [lesson, setLesson] = useState<LessonDetail | null>(null);
   115→  const [loading, setLoading] = useState(true);
   116→  const [isDark, setIsDark] = useState(false);
   117→  const [relatedSimulations, setRelatedSimulations] = useState<Simulation[]>([]);
   118→  const [showQuiz, setShowQuiz] = useState(false);
   119→
   120→  const isRTL = language === "ar";
   121→
   122→  useEffect(() => {
   123→    const fetchLesson = async () => {
   124→      try {
   125→        const data = await loadStaticData();
   126→        
   127→        let foundLesson: LessonDetail | null = null;
   128→        
   129→        for (const year of data?.academicYears || []) {
   130→          for (const subject of year.Subject || []) {
   131→            for (const unit of subject.Unit || []) {
   132→              const lesson = unit.Lesson?.find((l: any) => l.id === lessonId);
   133→              if (lesson) {
   134→                foundLesson = {
   135→                  ...lesson,
   136→                  Unit: { ...unit, Subject: subject }
   137→                };
   138→                
   139→                // Get related simulations
   140→                const simByLesson = getSimulationsByLessonId(lesson.id);
   141→                const simBySubject = getSimulationsBySubject(subject.nameAr);
   142→                const allSims = [...simByLesson];
   143→                simBySubject.forEach(sim => {
   144→                  if (!allSims.find(s => s.id === sim.id)) {
   145→                    allSims.push(sim);
   146→                  }
   147→                });
   148→                setRelatedSimulations(allSims);
   149→                
   150→                break;
   151→              }
   152→            }
   153→            if (foundLesson) break;
   154→          }
   155→          if (foundLesson) break;
   156→        }
   157→        
   158→        setLesson(foundLesson);
   159→      } catch (error) {
   160→        console.error("Error fetching lesson:", error);
   161→      } finally {
   162→        setLoading(false);
   163→      }
   164→    };
   165→    
   166→    fetchLesson();
   167→  }, [lessonId]);
   168→
   169→  const toggleTheme = () => {
   170→    setIsDark(!isDark);
   171→    document.documentElement.classList.toggle('dark');
   172→  };
   173→
   174→  const getTitle = () => {
   175→    if (!lesson) return "";
   176→    return language === "ar" ? lesson.titleAr : lesson.titleEn;
   177→  };
   178→
   179→  const getDescription = () => {
   180→    if (!lesson) return "";
   181→    return language === "ar" ? lesson.descriptionAr : lesson.descriptionEn;
   182→  };
   183→
   184→  const getIntroduction = () => {
   185→    if (!lesson) return "";
   186→    return language === "ar" ? lesson.introductionAr : lesson.introductionEn;
   187→  };
   188→
   189→  const getSummary = () => {
   190→    if (!lesson) return "";
   191→    return language === "ar" ? lesson.summaryAr : lesson.summaryEn;
   192→  };
   193→
   194→  const getObjectives = () => {
   195→    if (!lesson?.Objective) return [];
   196→    return lesson.Objective.sort((a, b) => a.order - b.order).map(obj => 
   197→      language === "ar" ? obj.textAr : obj.textEn
   198→    );
   199→  };
   200→
   201→  const getKeyConcepts = () => {
   202→    if (!lesson?.Concept) return [];
   203→    return lesson.Concept.sort((a, b) => a.order - b.order).map(concept => ({
   204→      term: language === "ar" ? concept.termAr : concept.termEn,
   205→      definition: language === "ar" ? concept.definitionAr : concept.definitionEn
   206→    }));
   207→  };
   208→
   209→  const getFormulas = () => {
   210→    if (!lesson?.Formula) return [];

   212→      formula: formula.formula,
   213→      explanation: language === "ar" ? formula.explanationAr : formula.explanationEn
   214→    }));
   215→  };

   218→    if (!lesson?.Example) return [];

   220→      question: language === "ar" ? example.questionAr : example.questionEn,
   221→      solution: language === "ar" ? example.solutionAr : example.solutionEn,
   222→      steps: language === "ar" ? example.stepsAr?.split('\n') : example.stepsEn?.split('\n')
   223→    }));
   224→  };
   225→
   226→  const getQuestions = () => {
   227→    if (!lesson?.Question) return [];
   228→    return lesson.Question.sort((a, b) => a.order - b.order);
   229→  };
   230→
   231→  return (
   232→    <div className={`min-h-screen bg-gradient-to-br from-slate-50 via-purple-50 to-pink-50 dark:from-slate-950 dark:via-purple-950 dark:to-pink-950 ${isRTL ? "rtl" : "ltr"}`} dir={isRTL ? "rtl" : "ltr"}>
   233→      {/* Header */}
   234→      <header className="sticky top-0 z-50 bg-white/80 dark:bg-slate-900/80 backdrop-blur-lg border-b border-slate-200 dark:border-slate-800">
   235→        <div className="max-w-7xl mx-auto px-4 py-4">
   236→          <div className="flex items-center justify-between">
   237→            <div className="flex items-center gap-4">
   238→              <Link href="/platform" className="flex items-center gap-2 hover:opacity-80 transition">
   239→                <img src="/logo.jpeg" alt="SmartEdu" className="w-10 h-10 rounded-xl object-cover" />
   240→                <span className="font-bold text-xl bg-gradient-to-r from-purple-600 to-pink-600 bg-clip-text text-transparent hidden sm:inline">
   241→                  {t("تعلم ذكي", "SmartEdu")}
   242→                </span>
   243→              </Link>
   244→            </div>
   245→
   246→            <div className="flex items-center gap-2">
   247→              {lesson?.Unit?.Subject && (
   248→                <Link href={`/platform/subject/${lesson.Unit.Subject.id}`}>
   249→                  <Button variant="ghost" size="sm" className="gap-2">
   250→                    <ArrowLeft className="w-4 h-4" />
   251→                    {t("العودة", "Back")}
   252→                  </Button>
   253→                </Link>
   254→              )}
   255→              <Button
   256→                variant="ghost"
   257→                size="sm"
   258→                onClick={toggleLanguage}
   259→                className="gap-2"
   260→              >
   261→                <Globe className="w-4 h-4" />
   262→                {language === "ar" ? "EN" : "عربي"}
   263→              </Button>
   264→              <Button
   265→                variant="ghost"
   266→                size="icon"
   267→                onClick={toggleTheme}
   268→              >
   269→                {isDark ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
   270→              </Button>
   271→            </div>
   272→          </div>
   273→        </div>
   274→      </header>
   275→
   276→      {/* Main Content */}
   277→      <main className="max-w-5xl mx-auto px-4 py-8">
   278→        {loading ? (
   279→          <div className="flex items-center justify-center py-20">
   280→            <div className="animate-spin w-12 h-12 border-4 border-purple-500 border-t-transparent rounded-full"></div>
   281→          </div>
   282→        ) : !lesson ? (
   283→          <div className="text-center py-20">
   284→            <h2 className="text-2xl font-bold text-slate-800 dark:text-white mb-4">
   285→              {t("الدرس غير موجود", "Lesson not found")}
   286→            </h2>
   287→            <Link href="/platform">
   288→              <Button>
   289→                {t("العودة للمنصة", "Back to Platform")}
   290→              </Button>
   291→            </Link>
   292→          </div>
   293→        ) : (
   294→          <>
   295→            {/* Breadcrumb */}
   296→            <nav className="flex items-center gap-2 text-sm text-slate-500 mb-6 flex-wrap">
   297→              <Link href="/platform" className="hover:text-purple-600 transition">
   298→                {t("المنصة", "Platform")}
   299→              </Link>
   300→              <ChevronRight className="w-4 h-4" />
   301→              {lesson.Unit?.Subject && (
   302→                <>
   303→                  <Link 
   304→                    href={`/platform/subject/${lesson.Unit.Subject.id}`} 
   305→                    className="hover:text-purple-600 transition"
   306→                  >
   307→                    {language === "ar" ? lesson.Unit.Subject.nameAr : lesson.Unit.Subject.nameEn}
   308→                  </Link>
   309→                  <ChevronRight className="w-4 h-4" />
   310→                </>
   311→              )}
   312→              <span className="text-slate-800 dark:text-white font-medium line-clamp-1">
   313→                {getTitle()}
   314→              </span>
   315→            </nav>
   316→
   317→            {/* Lesson Header */}
   318→            <motion.div
   319→              initial={{ opacity: 0, y: 20 }}
   320→              animate={{ opacity: 1, y: 0 }}
   321→              className="mb-8"
   322→            >
   323→              <div className="flex items-start gap-4 mb-4">
   324→                <div className={`w-12 h-12 rounded-xl flex items-center justify-center ${
   325→                  lesson.isFree 
   326→                    ? "bg-green-100 dark:bg-green-900/30" 
   327→                    : "bg-purple-100 dark:bg-purple-900/30"
   328→                }`}>
   329→                  {lesson.isFree ? (
   330→                    <Play className="w-6 h-6 text-green-600" />
   331→                  ) : (
   332→                    <Lock className="w-6 h-6 text-purple-600" />
   333→                  )}
   334→                </div>
   335→                <div className="flex-1">
   336→                  <h1 className="text-2xl md:text-3xl font-bold text-slate-800 dark:text-white mb-2">
   337→                    {getTitle()}
   338→                  </h1>
   339→                  <p className="text-slate-600 dark:text-slate-400 mb-3">
   340→                    {getDescription()}
   341→                  </p>
   342→                  <div className="flex items-center gap-4 text-sm text-slate-500">
   343→                    <div className="flex items-center gap-1">
   344→                      <Clock className="w-4 h-4" />
   345→                      {lesson.duration} {t("دقيقة", "min")}
   346→                    </div>
   347→                    {lesson.isFree && (
   348→                      <Badge className="bg-green-100 text-green-700 dark:bg-green-900/30 dark:text-green-400 border-0">
   349→                        {t("مجاني", "Free")}
   350→                      </Badge>
   351→                    )}
   352→                    {relatedSimulations.length > 0 && (
   353→                      <Badge className="bg-blue-100 text-blue-700 dark:bg-blue-900/30 dark:text-blue-400 border-0">
   354→                        <FlaskConical className="w-3 h-3 mr-1" />
   355→                        {relatedSimulations.length} {t("محاكاة", "simulations")}
   356→                      </Badge>
   357→                    )}
   358→                  </div>
   359→                </div>
   360→              </div>
   361→            </motion.div>
   362→
   363→            {/* Lesson Content Tabs */}
   364→            <Tabs defaultValue="content" className="space-y-6">
   365→              <TabsList className="grid grid-cols-2 md:grid-cols-6 gap-2 h-auto p-2 bg-white dark:bg-slate-800 rounded-xl">
   366→                <TabsTrigger value="content" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   367→                  <BookOpen className="w-4 h-4 mr-2" />
   368→                  {t("المحتوى", "Content")}
   369→                </TabsTrigger>
   370→                <TabsTrigger value="concepts" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   371→                  <Lightbulb className="w-4 h-4 mr-2" />
   372→                  {t("المفاهيم", "Concepts")}
   373→                </TabsTrigger>
   374→                <TabsTrigger value="formulas" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   375→                  <Target className="w-4 h-4 mr-2" />
   376→                  {t("القوانين", "Formulas")}
   377→                </TabsTrigger>
   378→                <TabsTrigger value="examples" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   379→                  <FileText className="w-4 h-4 mr-2" />
   380→                  {t("أمثلة", "Examples")}
   381→                </TabsTrigger>
   382→                <TabsTrigger value="resources" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   383→                  <Layers className="w-4 h-4 mr-2" />
   384→                  {t("المصادر", "Resources")}
   385→                </TabsTrigger>
   386→                <TabsTrigger value="quiz" className="rounded-lg data-[state=active]:bg-purple-100 data-[state=active]:text-purple-700 dark:data-[state=active]:bg-purple-900/30">
   387→                  <HelpCircle className="w-4 h-4 mr-2" />
   388→                  {t("اختبار", "Quiz")}
   389→                </TabsTrigger>
   390→              </TabsList>
   391→
   392→              {/* Content Tab */}
   393→              <TabsContent value="content">
   394→                <motion.div
   395→                  initial={{ opacity: 0, y: 10 }}
   396→                  animate={{ opacity: 1, y: 0 }}
   397→                  className="space-y-6"
   398→                >
   399→                  {getObjectives().length > 0 && (
   400→                    <Card className="border-0 shadow-md">
   401→                      <CardContent className="p-6">
   402→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   403→                          <Target className="w-5 h-5 text-purple-500" />
   404→                          {t("أهداف الدرس", "Lesson Objectives")}
   405→                        </h3>
   406→                        <ul className="space-y-2">
   407→                          {getObjectives().map((obj: string, i: number) => (
   408→                            <li key={i} className="flex items-start gap-3">
   409→                              <CheckCircle className="w-5 h-5 text-green-500 mt-0.5 flex-shrink-0" />
   410→                              <span className="text-slate-600 dark:text-slate-300">{obj}</span>
   411→                            </li>
   412→                          ))}
   413→                        </ul>
   414→                      </CardContent>
   415→                    </Card>
   416→                  )}
   417→
   418→                  {getIntroduction() && (
   419→                    <Card className="border-0 shadow-md">
   420→                      <CardContent className="p-6">
   421→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   422→                          <BookOpen className="w-5 h-5 text-purple-500" />
   423→                          {t("مقدمة الدرس", "Introduction")}
   424→                        </h3>

   427→                        </div>
   428→                      </CardContent>

   431→
   432→                  {getSummary() && (
   433→                    <Card className="border-0 shadow-md bg-gradient-to-r from-purple-50 to-pink-50 dark:from-purple-900/20 dark:to-pink-900/20">
   434→                      <CardContent className="p-6">
   435→                        <h3 className="text-lg font-bold text-slate-800 dark:text-white mb-4 flex items-center gap-2">
   436→                          <FileText className="w-5 h-5 text-purple-500" />
   437→                          {t("ملخص الدرس", "Summary")}
   438→                        </h3>
   439→                        <div className="prose dark:prose-invert max-w-none text-slate-600 dark:text-slate-300 whitespace-pre-wrap">
   440→                          {getSummary()}
   441→                        </div>
   442→                      </CardContent>
   443→                    </Card>
   444→                  )}
   445→                </motion.div>
   446→              </TabsContent>
   447→
   448→              {/* Concepts Tab */}
   449→              <TabsContent value="concepts">
   450→                <motion.div

   517→
   518→              {/* Examples Tab */}

   520→                <motion.div
   521→                  initial={{ opacity: 0, y: 10 }}
   522→                  animate={{ opacity: 1, y: 0 }}

   524→                  <Card className="border-0 shadow-md">
