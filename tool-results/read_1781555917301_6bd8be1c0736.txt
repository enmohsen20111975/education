  1561→// دالة للحصول على المحاكيات حسب المادة ورقم الوحدة
  1562→export function getSimulationsBySubjectAndUnit(subjectName: string, unitOrder: number): Simulation[] {
  1563→  // خريطة ربط المواد بأنواع المحاكيات
  1564→  const subjectTypeMap: Record<string, string[]> = {
  1565→    'الفيزياء': ['physics'],
  1566→    'Physics': ['physics'],
  1567→    'الكيمياء': ['chemistry'],
  1568→    'Chemistry': ['chemistry'],
  1569→    'الرياضيات': ['math'],
  1570→    'الرياضيات (1)': ['math'],
  1571→    'الرياضيات (2)': ['math'],
  1572→    'Mathematics': ['math'],
  1573→    'الأحياء': ['biology'],
  1574→    'Biology': ['biology'],
  1575→    'الجغرافيا': ['geography'],
  1576→    'Geography': ['geography'],
  1577→  };
  1578→
  1579→  const types = subjectTypeMap[subjectName] || [];
  1580→  
  1581→  // فلترة المحاكيات حسب النوع
  1582→  let filtered = simulations.filter(sim => types.includes(sim.type));
  1583→  
  1584→  // ترتيب المحاكيات حسب الوحدة (كل وحدة ليها محاكيات معينة)
  1585→  // الوحدة 1: المحاكيات الأولى، الوحدة 2: المحاكيات التالية، إلخ
  1586→  const simsPerUnit = 5; // عدد المحاكيات لكل وحدة
  1587→  const startIdx = Math.min((unitOrder - 1) * simsPerUnit, filtered.length - simsPerUnit);
  1588→  const endIdx = Math.min(startIdx + simsPerUnit, filtered.length);
  1589→  
  1590→  if (startIdx >= 0 && filtered.length > 0) {
  1591→    return filtered.slice(Math.max(0, startIdx), endIdx);
  1592→  }
  1593→  
  1594→  return filtered.slice(0, simsPerUnit);
  1595→}
